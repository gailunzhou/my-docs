    #include <stdio.h>
#include <string.h>
#include <stdbool.h>
#include <stdlib.h>

/*
工作人员管理系统

1.管理员

2.工作人员

*/

//1.管理员结构体
struct Manager
{
    int id;
    char *name;
    char *password;
};

//定义管理员数组
#define MANAGER_SIZE 5

struct Manager ms[MANAGER_SIZE];

//1.1管理员初始化
void Manager_Init(void);

void Show_Manager(void);//测试展示所有管理员

//1.2管理员登录
bool login(void);

//2.工作人员结构体
struct Worker
{
    int id;
    char *name;
    char *gender;
};

#define WORKER_SIZE 7

//工作人员数组
struct Worker workers[WORKER_SIZE];

//实际工作人员人数
int real_workernum;

//2.1工作人员初始化
void worker_init(void);

//3.1菜单
/*
printf("菜单\n");
printf("1.展示所有人员信息\n");
printf("2.查看指定人员信息\n");
printf("3.添加人员信息\n");
printf("4.修改指定人员信息\n");
printf("5.删除指定人员信息\n");
printf("6.退出系统\n");
printf("请输入功能编号 \n");
*/
void show_menu(void);

//3.2展示所有工作人员
void show_worker(void);

//3.2.1查看指定人员id
int search_workerid(void);
//3.2查看指定人员信息
void show_selectedwoker(void);

//3.3添加人员信息
void add_worker(void);

//3.4修改指定人员信息
void modifi_woker(void);

//3.5删除指定人员信息
void delet_worker(void);

int main(int argc, char const *argv[])
{
    Manager_Init();
    //Show_Manager();

    worker_init();

    //管理员登录
    int login_flag = login();
    int login_count = 0;

    while (login_flag != 1)
    {
        printf("输入的用户名或密码错误，请重新输入\n");
        login_count ++;
        if (login_count == 3)
        {
            printf("连续3次输入错误，当前禁止登录\n");
            break;
        }
        login_flag = login();
    }

    if (login_flag)
    {
        show_menu();
    }

    return 0;
}

void Manager_Init(void)
{
    struct Manager m1 = {1001 ,"a" ,"123"};
    struct Manager m2 = {1002 ,"b" ,"123"};
    struct Manager m3 = {1003 ,"c" ,"123"};
    struct Manager m4 = {1004 ,"d" ,"123"};
    struct Manager m5 = {1005 ,"e" ,"123"};

    ms[0] = m1;
    ms[1] = m2;
    ms[2] = m3;
    ms[3] = m4;
    ms[4] = m5;

}

bool login(void)
{
    printf("请输入用户名和密码(用空格隔开)\n");
    char username[20];
    char userpwd[20];
    scanf("%s %s" ,username ,userpwd);

    for (size_t i = 0; i < MANAGER_SIZE; i++)
    {
        if (!strcmp(username ,ms[i].name) && !strcmp(userpwd ,ms[i].password)) 
        {
            printf("登录成功！\n");
            return true; 
        }
    }
    return false;
}

void worker_init(void)
{
    struct Worker w1 = {2001 ,"A" ,"男"};
    struct Worker w2 = {2002 ,"B" ,"女"};
    struct Worker w3 = {2003 ,"C" ,"男"};
    struct Worker w4 = {2004 ,"D" ,"男"};
    struct Worker w5 = {2005 ,"E" ,"女"};

    workers[0] = w1;
    workers[1] = w2;
    workers[2] = w3;
    workers[3] = w4;
    workers[4] = w5;

    real_workernum = 5;
}

void show_worker(void)
{
    printf("id\t姓名\t性别\n");
    for (size_t i = 0; i < real_workernum; i++)
    {
        printf("%d\t%s\t%s\n" ,workers[i].id ,workers[i].name ,workers[i].gender);
    }
    show_menu();
}

int search_workerid(void)
{
    int input_id = 0;

    printf("请输入员工id:");
    scanf("%d" ,&input_id);

    for (size_t i = 0; i < real_workernum; i++)
    {
        if (input_id == workers[i].id)
        {
            return i;//返回查找的索引
        }
    }
    return -1;
}

void show_selectedwoker(void)
{
    int index = search_workerid();

    if (index == -1)
    {
        printf("无该id所对应员工，请重新尝试\n");
    }
    else
    {
        printf("id\t姓名\t性别\n");
        printf("%d\t%s\t%s\n" ,workers[index].id ,workers[index].name ,workers[index].gender);
    }
    show_menu();
}

void add_worker(void)
{
    char new_worker_name[10] = "";
    char new_worker_gender[10] = "";

    if (real_workernum == WORKER_SIZE)
    {
        printf("员工人数已满，无法添加，请进行其他操作\n");
    }
    else
    {   
        if (real_workernum == 0)
        {
            int new_worker_id = 2001;
            printf("请输入新员工的姓名和性别(用空格隔开):");
            scanf("%s %s" ,new_worker_name ,new_worker_gender);

            struct Worker new_worker = {new_worker_id ,new_worker_name ,new_worker_gender};
            workers[real_workernum] = new_worker;
        }
        else
        {
            int new_worker_id = workers[real_workernum - 1].id + 1;
            printf("请输入新员工的姓名和性别(用空格隔开):");
            scanf("%s %s" ,new_worker_name ,new_worker_gender);

            struct Worker new_worker = {new_worker_id ,new_worker_name ,new_worker_gender};
            workers[real_workernum] = new_worker;
        }
        real_workernum ++;

        printf("添加成功！\n");
        //打印添加后人员列表
        printf("添加后人员列表如下：\n");

        printf("id\t姓名\t性别\n");
        for (size_t i = 0; i < real_workernum; i++)
        {
            printf("%d\t%s\t%s\n" ,workers[i].id ,workers[i].name ,workers[i].gender);
        }    
    }

    show_menu();
}

void modifi_woker(void)
{
    int index = search_workerid();
    char str[10] = "";

    char new_name[10] = "";

    if (index == -1)
    {
        printf("无该id所对应员工，请重新尝试\n");
    }
    else
    {
        printf("员工信息为:\n");
        printf("id\t姓名\t性别\n");
        printf("%d\t%s\t%s\n" ,workers[index].id ,workers[index].name ,workers[index].gender);

        printf("请输入要修改的信息项：");
        scanf("%s" ,str);
        if (strcmp(str ,"姓名") == 0)
        {
            printf("请输入新的姓名：");
            scanf("%s" ,new_name);
            workers[index].name = new_name;
        }
        else if(strcmp(str ,"性别") == 0)
        {
            if (workers[index].gender == "男")
            {
                workers[index].gender = "女";
            }
            else
            {
                workers[index].gender = "男";
            }
        }

        printf("修改成功！改后信息为:\n");
        printf("id\t姓名\t性别\n");
        printf("%d\t%s\t%s\n" ,workers[index].id ,workers[index].name ,workers[index].gender);
    }
    show_menu();
}

void delet_worker(void)
{
    int index = 0;

    if (real_workernum == 0)
    {
        printf("暂无员工，请进行其他操作\n");
    }
    else
    {
        int index = search_workerid();
        if(index == -1)
        {
            printf("无该id所对应员工，请重新尝试\n");
        }
        else
        {
            for (size_t i = index; i < real_workernum - 1; i++)//除最后一项，用index的后项覆盖index项
            {
                workers[i] = workers[i+1];
            }
            //清空最后一项
            struct Worker clear = {0};
            workers[real_workernum - 1] = clear;

            //实际人员人数减1
            real_workernum --;

            //信息输出
            printf("删除信息成功！\n");
            //打印删除后人员列表
            printf("删除后人员列表如下：\n");

            printf("id\t姓名\t性别\n");
            for (size_t i = 0; i < real_workernum; i++)
            {
                printf("%d\t%s\t%s\n" ,workers[i].id ,workers[i].name ,workers[i].gender);
            }
        }
    }

    show_menu();
}

void show_menu(void)
{
    int order_num = 0;

    printf("\t菜单\n");

    printf("1.展示所有人员信息\n");
    printf("2.查看指定人员信息\n");
    printf("3.添加人员信息\n");
    printf("4.修改指定人员信息\n");
    printf("5.删除指定人员信息\n");
    printf("6.退出系统\n");

    printf("请输入功能编号: ");
    scanf("%d" ,&order_num);

    switch (order_num)
    {
    case 1://展示所有人员信息
        show_worker();
        break;
    case 2://查看指定人员信息
        show_selectedwoker();
        break;
    case 3://添加人员信息
        add_worker();
        break;
    case 4://修改指定人员信息
        modifi_woker();
        break;
    case 5://删除指定人员信息
        delet_worker();
        break;
    case 6://退出系统
        printf("退出系统\n");
        exit(EXIT_SUCCESS);
        break;

    default:
        printf("功能选项错误，系统异常\n");
        exit(1);
        break;
    }
}
