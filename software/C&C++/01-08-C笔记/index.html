    C语法基础

    1. 数组基础

    1.1 数组的创建与初始化

    数组定义：一个固定大小的相同类型元素的顺序集合

    创建（声明）语法：

    #include <stdio.h>

int mian()
{
    type array_name [size] ;    

    return 0;
}

    说明:

    • 其中的type就是数组中所要保存的数据的类型，同时也决定了数组的类型
    • 其中size必须是大于0的整数常量
    • 由于数组类型是由元素类型决定，因此必须保证元素类型都相同，否则会出现类型不匹配错误，如int类型的数组只能放int类型的数据
    • 编号规则：元素编号从0开始，最后一个元素编号为数组长度减1

    使用for循环可以更快的遍历数组：

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    //创建一个int类型长度为5的数组

    int myarray [5];//4 * 5 = 20 byte

    myarray[0] = 1;

    //数组的遍历

    int length = sizeof(myarray)/sizeof(myarray[0]);//计算数组里有几个元素

    for (size_t i = 0; i < length; i++)
    {
        printf("myarray[%d] = %d \n" ,i ,myarray[i]);
    }

    //结果：定义的数组是局部变量未初始化，数组的元素会有脏数据

    return 0;
}

    关于初始化：

    数组如果没有进行初始化，由于整体数组是一个局部变量，因此其中的元素会有脏数据

    例如：

    #include <stdio.h>

//使用宏定义的方式来计算数组有几个元素
#define LENGTH(array) (sizeof(array)/sizeof(array[0]))

int main(int argc, char const *argv[])
{
    //数组初始化
    //1.完全初始化

    double myarray1[3] = {9.1 ,7.7 ,5.3};

    for (size_t i = 0; i < LENGTH(myarray1); i++)
    {
        printf("myarray1[%d] = %.2f\t" ,i ,myarray1[i]);
    }

    printf("\n");

    //2.不完全初始化 ,未初始化的元素有脏数据

    int myarray2[5] = {2 ,3 ,6};

    for (size_t i = 0; i < LENGTH(myarray2); i++)
    {
        printf("myarray2[%d] = %d\t" ,i ,myarray2[i]);
    }

    printf("\n");

    //3.不固定长度初始化
    int myarray3[] = {2 ,3 ,6 ,12 ,6 ,8 ,4546 ,9 ,89 ,56 ,12};

    for (size_t i = 0; i < LENGTH(myarray3); i++)
    {
        printf("myarray3[%d] = %d\t" ,i ,myarray3[i]);
    }

    printf("\n");

    return 0;
}

    1.2 字符串的基础操作

    需要包含字符串相关的头文件：

    #include <string.h>

    1.2.1 输出字符串

    使用**puts()**函数

    #include <stdio.h>
#include <string.h>

int main(int argc, char const *argv[])
{
    //字符串本质就是以'\0'结尾的字符型数组

    char str1[] = "abcdefg";

    //输出字符串函数
    puts(str1);

    return 0;
}

    1.2.2 输出字符串内容长度

    使用**strlen()**函数

    #include <stdio.h>
#include <string.h>

int main(int argc, char const *argv[])
{
    //字符串本质就是以'\0'结尾的字符型数组

    char str1[] = "abcdefg";

    //输出字符串字符长度函数
    printf("str1有%d个内容 \n",strlen(str1));

    return 0;
}

    1.2.3 字符串追加操作

    1.2.3.1 **strcat()**函数

    **strcat()**函数不限制所追加的字符串长度，容易导致数组越界

    例如：

    #include <stdio.h>
#include <string.h>

int main(int argc, char const *argv[])
{
    char str1[10] = "a";
    char str2[30] = "bcdefghijklomn";

    //strcat不限制追加长度，会导致数组越界
    puts(strcat(str1 ,str2));//可以看到输出的字符串长度已经超过了str1的规定长度

    return 0;
}

    1.2.3.2 **strncat()**函数

    strncat()函数可以通过指定参数n，来指定要追加的长度，可以避免数组越界

    例如：

    #include <stdio.h>
#include <string.h>

int main(int argc, char const *argv[])
{
    char str1[10] = "a";
    char str2[30] = "bcdefghijklomn";

    //strncat可以限制追加的长度: 字符串数组的总长度-'\0'位-已使用长度
    char str3[10] = "a";
    puts(strncat(str3 ,str2 ,(sizeof(str3) - 1 -strlen(str3))));

    return 0;
}

    1.2.4 字符串复制操作

    1.2.4.1 **strcpy()**函数

    **strcpy()函数不会限制要复制字符串的长度，和strcat()**类似，会导致数组越界操作

    例如：

    #include <stdio.h>
#include <string.h>

int main(int argc, char const *argv[])
{
    char str4[10] = "a";
    char str5[30] = "bcdefghijklomn";

    //strcapy不限制复制长度，会导致数组越界
    puts(strcpy(str4 ,str5));//可以看到输出的字符串长度已经超过了str4的规定长度

    return 0;
}

    1.2.4.2 **strncpy()**函数

    strncpy()函数可以根据参数n，设置要复制的字符串长度

    例如：

    #include <stdio.h>
#include <string.h>

int main(int argc, char const *argv[])
{
    char str5[30] = "bcdefghijklomn";

    //strncpy可以限制复制的长度: 字符串数组的总长度-'\0'位
    char str6[10] = "a";
    puts(strncpy(str6 ,str5 ,(sizeof(str6) - 1)));

    return 0;
}

    1.2.5 字符串比较操作

    1.2.5.1 **strcmp()**函数

    strcmp()是将两个字符串所有都比较，如果前者中的字符大于后者中的字符，返回1，如果全相同，返回0，如果前者小于后者则返回-1

    例如:

    #include <stdio.h>

int mian(int argc, char const *argv[])
{
    char str7[10] = "aaabd";
    char str8[30] = "aaacd";

    printf("str7 和 str8 比较结果为 :%d \n" ,strcmp(str7 ,str8));//可以看到输出为-1，因为'b'在ASCII码中小于'c'   

    return 0;
}

    如果要比较两个字符串是否相等可以用这种写法：

    if(!strcmp(str1,str2))
{
    ...
}
else
{
    ...
}
//因为strcmp两个相等的话返回0，0在bool值里面是“假”，因此直接取反表示“真”，即“如果返回是‘假’，那么就执行”

    1.2.5.2 **strncmp()**函数

    strncmp()是可以通过参数n指定要比较的前n位，其返回逻辑和返回结果和**strcmp()**相同

    例如：

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    char str7[10] = "aaabd";
    char str8[30] = "aaacd";

    printf("str7 和 str8 的前3位比较结果为 :%d \n" ,strncmp(str7 ,str8 ,3));//可以看到输出0，因为前三位都是'a'
    printf("str7 和 str8 的前4位比较结果为 :%d \n" ,strncmp(str7 ,str8 ,4));//可以看到输出-1
    return 0;
}

    1.2.6 字符串查找操作

    1.2.6.1 strchr()函数

    strchr()是查找字符第一次出现的位置，返回指向该位置的指针，因为是指针，说明其返回的是一个地址，因此要创建一个char 类型的指针用来存储返回值

    例如：

    #include <stdio.h>
//查找第一次出现'e'的位置
int main(int argc, const char *argv[])
{
    char str9[30] = "aaabbbdeffffff";

    char *res;
    res = strchr(str9 ,'e');
    printf("str9第一次出现'e'的位置是%s \n" ,res);//str9第一次出现'e'的位置是effffff

    return 0;
}

    1.2.6.2 strrchr()函数

    strrchr()是查找字符最后一次出现的位置，返回指向该位置的指针，因为是指针，说明其返回的是一个地址，因此要创建一个char 类型的指针用来存储返回值

    例如：

    #include <stdio.h>
//查找最后一次出现'b'的位置
int main(int argc, const char *argv[])
{
    char str9[30] = "aaabbbdeffffff";

    char *res;

    res = strrchr(str9 ,'b');
    printf("str9最后一次出现'b'的位置是%s \n" ,res);//str9最后一次出现'b'的位置是bdeffffff

    return 0;
}

    1.2.7 字符串分解操作

    1.2.7.1 **strtok()**函数

    **strtok()**可以将字符串按指定分隔符进行拆分，第一次调用传入字符串和分隔符，后续调用传入 NULL 和分隔符，直到返回 NULL 表示分解结束

    其返回的是指向下一个标记的指针，如果没有更多标记，则返回NULL

    因此需要创建一个char类型的指针用来存储返回值

    例如：

    #include <stdio.h>
//将www.ccchiji.com按照.进行分隔
int main(int argc ,const char *argv[])
{
    char str[] = "www.ccchiji.com";
    char c[] = ".";

    char *token;

    //第一次调用传入字符串和分隔符
    token = strtok(str ,c);

    //后续调用传入NULL和分隔符，直到返回NULL表示分解结束
    while(token != NULL)
    {
        printf("%s\n",token);

        token = strtok(NULL ,c);
    }
    /*
    打印出的结果应该是：
    www
    ccchiji
    com
    */
    return 0;
}

    说明：

    为什么第一次是传入目标字符串和分隔符，而后续是传入NULL和分隔符？

    strtok()函数的拆分原理是：找到目标字符串中的分隔符，将其替换为结束符'\0'，然后内部保存下一次拆分的起始位置

    拆分过程：

    第一次拆分：传入目标字符串和分隔符，是为了告诉要拆分哪一个字符串

    第二次及后续拆分：传入NULL和分隔符，是为了告诉不要重新拆分，从上一次拆分结束后的位置继续拆分，如果是还是传入目标字符串和分隔符，这其实是让函数重新拆分

    该函数的分隔符可以不是单个，因为传入的是一个字符串，可以是多个

    例如：

    #include <stdio.h>

int main(int argc ,const char *argv[])
{
    char str[] = "a,b;c-d";
    char c[] = ",;-";

    char *token;

    printf("原始字符串：%s \n",str);

    token  = strtok(str ,c);

    while(token != NULL)
    {
        printf("拆分后字符串：%s \n" ,token);

        token  = strtok(NULL ,c);
    }   
    /*
    打印结果为：
    原始字符串：a,b;c-d
    拆分后字符串：a
    拆分后字符串：b
    拆分后字符串：c
    拆分后字符串：d    
    */

    return 0;
}

    2. 指针

    2.1 指针基础

    指针存储的是：所指向数据的基地址（起始地址）

    关于指针变量：将所指向的数据的基地址，在内存中开辟一块地方，存起来，因此指针变量在内存中也有一个地址

    指针变量的基础运用：

    #include <stdio.h>

int main(int argc ,const char *argv[])
{
    //普通变量
    int age = 18;
    //指针变量
    int *p = &age;

    //输出age的原始地址
    printf("age的值为 %d age的原始地址为：%p \n" ,age ,p);

    //直接修改变量自身，不会修改其地址
    age = 20;
    printf("age的值为 %d age的原始地址为：%p \n" ,age ,p);

    //直接修改变量的地址所存储的内容也可以实现修改变量的目的(指针的解引用)
    *p = 22;
    printf("age的值为 %d age的原始地址为：%p \n" ,age ,p);

    return 0;
}

    指针变量的大小（其所占用的内存）只和系统的位数有关，和其所指向的数据类型无关

    可通过sizeof()函数查看

    #include <stdio.h>

int main(int argc ,const char *argv[])
{
    int *p = NULL;

    printf("指针变量所占内存为：%zu \n" ,sizeof(p));//64位系统则打印为 8 32位系统则是 4

    return 0;
}

    内存是从低地址向高地址累计，系统中每一个字节(byte)都有一个地址，因此可以通过指针的方式查看其所存储类型数据所占用的内存大小

    例如：

    #include <stdio.h>

int main(int argc ,const char *argv[])
{
    int a = 10;
    int b = 11;

    int *pa = &a;
    int *pb = &b;

    printf("a的基地址为：%p \n" ,pa);//00000004da7ffaf0
    printf("b的基地址为: %p \n" ,pb);//00000004da7ffaec

    /*
    根据上面可知b和a地址之间为:
    a地址:  00000004da7ffaf0
            00000004da7ffaff
            00000004da7ffafe
            00000004da7ffafd
    b地址:  00000004da7ffafc

    地址之间相差4 ，说明隔了4个byte，也就是b作为int型的数据占了4个byte
    */

    return 0;
}

    2.2 指针的加减

    本质：对所指数据的地址按照数据类型所占内存大小进行移动

    关键：

    由于指针加减是对一个已知指向类型的指针进行地址移动操作

    但移动后所指地址内存储的数据类型和原来指针所指的数据类型是否一致？这是未知的

    因此一般都是在已知范围内进行移动

    如果随意移动，指针移动的位数只会按照自己所指向的类型所占byte数进行移动

    比如： 一个int类型的指针int *p = \&a; p+1是向高地址移动4位（int型） 假设当前仅知道一个int 型的a变量，那么该变量地址向上4字节地址所存的是否是int型数据？这是未知的 因此这种错误的运算会导致野指针的出现，强烈不建议

    2.2.1加法

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    //指针运算：加法
    int a = 10;
    int b = 11;
    int c = 12;

    int *pa = &a;
    int *pb = &b;
    int *pc = &c;

    printf("pa所指向的地址是 %p\tpb所指向的地址是%p\tpc所指向的地址是%p\t \n" ,pa ,pb ,pc);
    //进行指针加法时，操作的是被加的那个指针，加1，代表地址向高地址移动，而其移动的位数，仅取决于 被加指针所指的数据类型占用的byte数
    printf("pc+0所指向的地址是 %p 内容是 %d\n" ,(pc+0) ,*(pc+0));
    printf("pc+1所指向的地址是 %p 内容是 %d\n" ,(pc+1) ,*(pc+1));
    printf("pc+2所指向的地址是 %p 内容是 %d\n" ,(pc+2) ,*(pc+2));

    double d = 13.0;
    double e = 14.0;
    double f = 15.0;

    double *pd = &d;
    double *pe = &e;
    double *pf = &f;

    printf("pd所指向的地址是 %p\tpe所指向的地址是%p\tpf所指向的地址是%p\t \n" ,pd ,pe ,pf);
    //进行指针加法时，操作的是被加的那个指针，加1，代表地址向高地址移动，而其移动的位数，仅取决于 被加指针所指的数据类型占用的byte数
    printf("pf+0所指向的地址是 %p 内容是 %.2f\n" ,(pf+0) ,*(pf+0));
    printf("pf+1所指向的地址是 %p 内容是 %.2f\n" ,(pf+1) ,*(pf+1));
    printf("pf+2所指向的地址是 %p 内容是 %.2f\n" ,(pf+2) ,*(pf+2));

    return 0;
}

    2.2.2 减法

    和加法同理

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    int a = 10;
    int b = 11;
    int c = 12;

    int *pa = &a;
    int *pb = &b;
    int *pc = &c;

    //指针运算：减法

    //进行指针减法时，操作的是被减的那个指针，减1，代表地址向低地址移动，而其移动的位数，仅取决于 被加指针所指的数据类型占用的byte数
    printf("pa-0所指向的地址是 %p 内容是 %d\n" ,(pa-0) ,*(pa-0));
    printf("pa-1所指向的地址是 %p 内容是 %d\n" ,(pa-1) ,*(pa-1));
    printf("pa-2所指向的地址是 %p 内容是 %d\n" ,(pa-2) ,*(pa-2));

    return 0;
}

    3. 两种特殊指针

    3.1 野指针

    什么是野指针：指针保存的地址是非法内存地址，其值不是程序分配的值，无法使用且难以判断

    野指针是怎么产生的：

    • 局部变量指针没有初始化
    • 指针指向的变量在指针之前被销毁
    • 使用已经释放的指针
    • 类型错误转换
    • 进行错误的指针运算

    关于什么是已经释放的指针：

    曾经指向一块有效内存，但该内存已被手动释放或自动回收（如栈内存出作用域）后，指针本身并未被置空，仍然保留着原来的内存地址。

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    //指针变量（局部变量）未赋值时，其会存一个地址，而内容未知，此时就是野指针
    int *p1;

    return 0;
}

    3.2 空指针

    什么是空指针？

    指向地址是0x0000.....000的指针，其内容为NULL

    空指针没有危险，一般用来防止野指针

    例如:

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    //一般为了避免野指针，会给未知内容的指针赋值NULL，使其变为空指针，方便后面赋值有具体的内容
    int *p = NULL;
    if (p == NULL)
    {
        printf("p是空指针");
    }

    return 0;
}

    4. 数组指针

    什么是数组指针：指向整个数组的指针

    数组指针的关键：数组指针的长度和数据类型要和原数组严格匹配

    定义格式：

    type (* pointer_name)[];

    和基本指针的区别：

    #include <stdio.h>

//计算数组长度的宏
#define LENGTH(array) (sizeof(array)/sizeof(array[0]))

int main(int argc, char const *argv[])
{
    //数组的指针指向的是数组整体
    //数组
    int myarray[9] = {1 ,2 ,3 ,4 ,5 ,6 ,7 ,8 ,9};

    //遍历输出
    for (size_t i = 0; i < LENGTH(myarray); i++)
    {
        printf("myarray[%zu] = %d " ,i ,myarray[i]);
    }
    printf("\n");

    //数组指针
    int (*myarrayp) [9] = &myarray;//其实就是一个指向int [9]类型的名称为myarrayp的指针
    printf("数组指针所指向地址为%p \n" ,myarrayp);

    //和普通指针的区别可以通过：指向的值  指针加法所跳过的字节数  进行区分

    //普通指针
    int *myarrayp2 = myarray;//指向myarray[0]的int类型的名称为myarrayp2的指针
    printf("普通指针所指向地址为%p \n" ,myarrayp2);

    //1.指向的值
    printf("普通指针myarrayp2指向的地址所保存的内容为：%d \n" ,*myarrayp2);
    printf("数组指针myarrayp指向的地址所保存的内容为：%d \n" ,*myarrayp[0]);

    //2.指针加法所跳过的字节数
    printf("普通指针myarrayp2所占内存大小为:%zu,其+1后所跳过的字节数为4 \n" ,sizeof(myarrayp2));
    printf("数组指针myarrayp所占内存大小为:%zu,其+1后所跳过的字节数为 4*9 = 36 \n" ,sizeof(myarrayp));

    printf("\n");

    printf("数组指针myarrayp之前的所指地址为:%p\t+1后所指地址为:%p \n",myarrayp ,myarrayp+1);//00000068711ffca0  00000068711ffcc4 相差36

    //补充：数组名也具有指针的作用，指向数组的基地址
    printf("数组名的也指向数组的基地址：%p \n" ,myarray);

    return 0;
}

    5. 指针数组

    什么是指针数组：内容全是指针的一个数组

    定义格式：

    type *pointer_name[];

    基础使用：

    #include <stdio.h>

#define LENGTH(array) (sizeof(array)/sizeof(array[0]))

int main(int argc, char const *argv[])
{
    int a = 555;
    int b = 666;
    int c = 777;

    //普通数组
    int array[] = {a ,b ,c};

    int *pa = &a;
    int *pb = &b;
    int *pc = &c;

    //指针数组

    int * p[] = {pa ,pb ,pc};

    for (size_t i = 0; i < LENGTH(array); i++)
    {
        printf("array[%zu] = %d\t" ,i ,array[i]);
    }

    printf("\n");

    for (size_t i = 0; i < LENGTH(p); i++)
    {
        printf("指针数组p[%zu]的地址是%p ,内容是%d\t" ,i ,p[i] ,*p[i]);
    }


    return 0;
}
