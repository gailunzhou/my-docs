    C 语法基础

    1.输入输出

    1.1 单个输入输出

    1.1.1 getchar()函数

    getchar()函数可以从输入设备（键盘）中逐个单次获取输入的字符类型数据，并将其从缓冲区中清除

    例如：

    #include <stdio.h>

int main()
{
    //提示语
    printf("请输入一个字符，并用回车结束 \n");
    char c1;
    c1 = getchar();
    printf("输入的字符是 %c ,其在ASCII码表中对应的是 %d \n" ,c1 ,c1);  

    return 0;
}

    1.1.2 putchar()函数

    putchar()函数可以将字符型数据逐个单次输出到标准输出设备（屏幕）

    例如：

    #include <stdio.h>

int main()
{
    //输出一个字符
    putchar('c');

    return 0;
}

    2. 多个输入输出

    2.1 scanf()函数

    scanf()函数可以一次性从缓冲区中获取多个、多种类型的数据，但对输入格式有严格要求

    例如：

    #include <stdio.h>

int main(int argc, char const *argv[])
{
   //提示语句
   //整型数据
   printf("请输入两个正整数，两数之间用一个空格隔开，例如：1 2 \n");
   int num1 , num2 ;
   scanf("%d %d" ,&num1 ,&num2);
   printf("输入的两个数是：%d 和 %d \n" ,num1 ,num2);

   //实型数据
    printf("请输入一个正整数，一个两位小数，之间用空格隔开，例如：1 2.22 \n");
    int num3 = 0;
    double num4 = 0.0;
    scanf("%d %lf" ,&num3 ,&num4);
    printf("输入的两个数是：%d 和 %.2f \n" ,num3 ,num4);

    //字符类型
    printf("请输入一个正整数，一个两位小数，一个字符，之间用空格隔开，例如：1 2.22 c \n");
    int num5 = 0;
    double num6 = 0.0;
    char c1 = '\0';
    scanf("%d %lf %c" ,&num5 ,&num6 ,&c1);
    printf("输入的是：%d 和 %.2f 和 %c \n" ,num5 ,num6 ,c1);

    return 0;
}

    注意：

    scanf()函数在以下情况会出现结果错误的问题：

    假设上一次输入结尾缓冲区是回车（'\n'）时， 同时下一次只想要读取一个字符， 会将缓冲区中的换行符读取进去而不是等待用户输入

    解决方式：

    使用getchar()函数对缓冲区进行逐个读取并丢弃所读取到的内容，可以实现清空缓冲区的作用，可利用该特性实现循环接收输入

    例如：

    #include <stdio.h>

int main(int argc, char const *argv[])
{
   printf("请输入两个正整数，两数之间用一个空格隔开，例如：1 2 \n");
   int num7 = 0, num8 = 0;
   scanf("%d %d" ,&num7 ,&num8);//这里结尾是有一个换行符的
   printf("输入的两个数是：%d 和 %d \n" ,num7 ,num8);

   //清空缓冲区：将缓冲区的内容逐个读取并丢弃直到读到换行符，再将换行符丢弃
   while (getchar() != '\n');

   printf("请输入两个字符，用空格隔开，例如：c a \n");
   char c2 ,c3;
   scanf("%c %c" ,&c2 ,&c3);
   printf("输入的两个字符是：%c 和 %c \n" ,c2 ,c3);

   return 0;
}

    注意：

    scanf()函数是有返回值的，返回从输入设备中获得的字符个数

    例如：

    #include <stdio.h>

int main()
{
    printf("请输入2个字符，字符之间用空格隔开，例如：a b \n");
    char c1 ,c2 ;
    int input_num = 0;
    input_num = scanf("%c %c" ,&c1 ,&c2);
    printf("输入的字符分别是：%c %c 一共 %d 个 \n" ,c1 ,c2 ,input_num);

    return 0;
}

    3. 类型转换

    3.1 隐式转换

    该方式的转换可在以下两种情况看到：

    3.1.1 计算

    该情况下是默认将小的类型转为大的类型，int是4byte，double是8byte，double所表示的范围更大

    根据此可知，short 和 char → int 类型 ，int 类型 → unsigned 类型 ，unsigned 类型 → long 类型，long 类型 → double 类型，而 float 类型 → double 类型，以上所有都可以转为 long double 类型

    但大的类型不能转换为小的类型

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    int a = 5;
    double b = 10.0;
    printf("a = 5 ,b = 10.0 ,a * b 的结果为 %.2f \n" ,a * b);

    return 0;
}

    3.1.2 赋值

    该情况下是自动将等号右侧的数据转为左侧类型的数据，因此可能存在精度丢失（如浮点数转整数）

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    double b = 10.0;
    int c = b;
    printf("b = 10.0 ,c = %d ,b = %.2f \n" ,c ,b);

    return 0;
}

    注意：

    该方式不会影响原值

    3.2 强制类型转换

    通过在数据前面加上小括号，小括号内写入目的类型的方式实现类型转换，例如：

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    double hight = 200.12;
    printf("hight原值为200.12，强制转换为int型后结果为：%d \n" ,(int)hight);//200

    return 0;
}

    注意：

    该方式也会存在精度丢失问题

    细节说明:

    • 类型转换主要涉及到运算，而char和short类型参与运算都要先转为int类型
    • 所有的浮点型计算都以双精度进行，即使只有float类型的数据也要先都转为double类型进行运算
    • 可以自动转换的类型一定可以强制转换，可以强制转换的类型不一定可以自动转换

    4. 算数运算符

    按照数学运算规则运算

    #include <stdio.h>

//算数运算符

int main(int argc, char const *argv[])
{
    int num1 = 5 ,num2 = 7;
    printf("num1 = %d ,num2 = %d \n" ,num1 ,num2);

    //+
    printf("num1 + num2 = %d \n" ,num1 + num2);
    //-
    printf("num1 - num2 = %d \n" ,num1 - num2);
    //*
    printf("num1 * num2 = %d \n" ,num1 * num2);
    // /
    printf("num1 / num2 = %d \n" ,num1 / num2);
    //如果整数和实数进行相除，整数会自动转换为实数，结果按结果数据类型确定
    printf("3 / 1.50 = %.2f \n" ,3 / 1.50);//结果类型是实型
    //%
    printf("num1 %% num2 = %d \n" ,num1 % num2);

    return 0;
}

    其中取余运算结果的正负，是由被取余数的正负所决定：

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    //取余结果的正负按照被取余的正负决定
    printf("-10 %% 3 = %d \n" ,-10 % 3);//-1
    printf("10 %% -3 = %d \n" ,10 % -3);//1

    return 0;
}

    取余运算中，运算的数据必须都是整数

    5. 关系运算符

    也叫比较运算符，其运算结果是bool类型数据

    #include <stdio.h>

//比较运算符 ,运算结果是bool类型数据

int main(int argc, char const *argv[])
{
    int num1 = 3 ,num2 = 5;

    /* code */
    //1.<
    printf("3 < 5为 %d \n", 3 < 5);

    //2.<=
    printf("3 <= 5为 %d \n", 3 <= 5);

    //3.>
    printf("3 > 5为 %d \n", 3 > 5);

    //4.>=
    printf("3 >= 5为 %d \n", 3 >= 5);

    //5.==
    printf("3 == 5为 %d \n", 3 == 5);

    //6.!=
    printf("3 != 5为 %d \n", 3 != 5);

    return 0;
}

    说明：

    • 关系运算中 < 、<= 、>、>= 的优先级高于 != 、==
    • 关系运算的优先级低于算数运算
    #include <stdio.h>

//比较运算符 ,运算结果是bool类型数据

int main(int argc, char const *argv[])
{
    //关系运算中 < <= > >= 优先级高于 != ==
    printf("3 <= 5 != 4 >=6 结果为 %d \n" ,3 <= 5 != 4 >=6);//1

    //优先级低于算数运算
    printf("3+5 > 2*2 的结果为 %d \n" ,3+5 > 2*2);//8 > 4 结果为 1

    return 0;
}

    6. 赋值运算符

    分为直接赋值和混合赋值

    6.1 直接赋值

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    //直接赋值
    int num3 = 12;
    num3 = 5;
    printf("num3 = %d \n" ,num3);

    return 0;
}

    6.2 混合赋值

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    /* code */
    int num1 = 10 ,num2 = 16;

    //1. +=     num1 = num1 + 10
    printf("num1 += 10 结果为 %d \n" ,num1 += 10);

    //2. -=     num1 = num1 - 10
    printf("num1 -= 10 结果为 %d \n" ,num1 -= 10);

    //3. *=     num1 = num1 * 10
    printf("num1 *= 10 结果为 %d \n" ,num1 *= 10);

    //4. /=     num1 = num1 / 10
    printf("num1 /= 10 结果为 %d \n" ,num1 /= 10);

    //5. %=     num1 = num1 % 10
    printf("num1 %%= 10 结果为 %d \n" ,num1 %= 10);

    return 0;
}

    混合赋值中，等号右侧可以是一个表达式而不是一个数:

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    /* code */
    int num1 = 10 ,num2 = 16;
    //右侧可以是一个表达式，当成一个整体
    printf("num1 += num2 - 6 结果为 %d \n" ,num1 += num2 - 6);//20

    return 0;
}

    比较运算符的优先级高于赋值运算符:

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    int num1 = 10 ,num2 = 16;

    //比较运算符优先级高于赋值运算符
    int num4 = num1 > num2;
    printf("num4 = num1 > num2   num4 = %d \n" ,num4);//0

    return 0;
}

    关于赋值运算要明确的是:

    赋值运算仅适用于变量，不适用于常量

    7. 逻辑运算符

    7.1 基本逻辑运算

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    int a = 5 ,b = 10;
    //逻辑运算符

    //1. && 与运算 
    printf("5 && 8 的结果为 %d \n" ,5 && 8);//结果为1

    //2. || 或运算
    printf("5 || 8 的结果为 %d \n" ,5 || 8);//结果为1

    //3. ! 非运算
    printf("!5 的结果为 %d \n" ,!5);//结果为0

    return 0;
}

    7.2 短路运算

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    int a = 5 ,b = 10;

    //1. 与运算的短路运算 ，遇到false则直接为false
    printf("a = 5,b = 10 ,!a && !b 的结果为 %d ,a为%d ,b为%d \n" ,!a && !b ,a ,b);

    //2. 或运算的短路运算 ，遇到true则直接为true
    printf("a = 5,b = 10 ,a || !b 的结果为 %d ,a为%d ,b为%d \n" ,a || !b ,a ,b);

    return 0;
}

    注意：

    与运算的短路运算中，&&号左侧的若为false，右侧的表达式不执行，建议将易假值放在左侧

    同理位运算的短路运算中，||号左侧的若为true，右侧的表达式不执行，建议将易真值放在左侧

    因此是可以加快程序运行

    7.3 运算优先级

    逻辑运算符的运算优先级 < 关系运算符 < 算数运算符

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    int a = 5 ,b = 10;

    //逻辑运算符优先级低于关系运算符
    printf("5 > 8 && 3 < 2 的结果为 %d \n" ,5 > 8 && 3 < 2);
    //也低于算数类运算
    a = 0; 
    b = 2;
    printf("a = 0 ,b = 2 ,a++ && b-- 的结果为 %d ,a为%d ,b为%d \n" ,a++ && b-- ,a ,b);

    return 0;
}

    8. 逗号运算符

    逗号运算符的运算优先级最低，因此在运算过程中一般需要加小括号,，而其取值是小括号最右侧的结果数据：

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    //逗号运算符
    //表示连续，在赋值中赋值结果为最右侧的结果，同时必须要加小括号
    int a = 5, b = 10;
    int res = (a++ , ++a ,b-- ,a+b);//16
    printf("res = %d ,a = %d ,b = %d \n" ,res ,a ,b);//16  7  9

    return 0;
}

    9. 三目运算符

    基本使用语法：

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    //三目运算
    //基本语法: 表达式1 ？ 表达式2 ：表达式3 ; 表达式1成立则执行表达式2，否则执行表达式3
    int a = 5 ,b = 4;
    int res = a > 4 ? a++ : b-- ;
    printf("a = 5 ,b = 4 ,a > 4 ? a++ : b-- 的结果为 %d ,a为 %d ,b为%d \n" ,res ,a ,b);

    return 0;
}

    注意1：

    三目运算中，表达式类型必须可兼容，即要么都为统一类型，要么类型之间可隐式转换

    #include <stdio.h>

int main()
{
    int num = 110;
    int res = num > 0 ? 100 : "错误"; // 编译报错：类型不匹配

    return 0;
}

    注意2：

    三目嵌套运算中，最好为内部的三目运算加上括号，防止歧义

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    //嵌套运算
    //将三个表达式中的表达式2或表达式3替换为一个三目运算
    int c = 6 ,d = 12;
    a = 5 ,b = 4;
    res = a < c ? d++ : c > b ? a-- : b++;
    printf("a = 5 ,b = 4 ,c = 6 ,d = 12 ,a < c ? d++ : c > b ? a-- : b++ 的结果为 %d ,a = %d ,b = %d ,c = %d ,d = %d \n", res, a, b, c, d);
    /*
    嵌套运算极力建议为内层三目增加括号以避免歧义
    即上述嵌套运算可能存在该种理解：
    res = (a < c ? d++ : c > b) ? a-- : b++;
    */
    c = 6 ,d = 12 ,a = 5 ,b = 4;
    res = a < c ? d++ : (c > b ? a-- : b++);
    printf("a = 5 ,b = 4 ,c = 6 ,d = 12 ,a < c ? d++ : (c > b ? a-- : b++) 的结果为 %d ,a = %d ,b = %d ,c = %d ,d = %d \n", res, a, b, c, d);

    return 0;
}

    三目运算的优先级：

    • 三目运算 > 赋值运算符
    • 三目运算符 < 算数、逻辑、关系运算符

    10. 自增自减运算

    关键点：

    • 前置自增/减运算，是先改值，后使用
    • 后置自增/减运算，是先使用，后改值
    #include <stdio.h>

int main(int argc, char const *argv[])
{
    int a = 5 ,b = 6;
    //自增自减运算
    //1.自增
    //1.1 前置自增  :先改值，后使用
    printf("++a的结果为 %d ,a = %d \n" ,++a ,a);//6 6
    //1.2 后置自增  :先使用，后改值
    a = 5;
    printf("a++的结果为 %d ,a = %d \n" ,a++ ,a);//5 6

    //2.自减
    //2.1 前置自减  :先改值，后使用
    printf("--b的结果为 %d ,b = %d \n" ,--b ,b);//5 5
    //2.2 后置自减  :先使用，后改值
    b = 6;
    printf("b--的结果为 %d ,b = %d \n" ,b-- ,b);//6 5

    //自增自减不能用于常量
    const int AGE = 18;
    //不能++AGE AGE++ --AGE AGE--

    return 0;
}

    11. 位运算

    基本位运算：

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    //位运算
    //1. 位与运算(&) 都为1时才为1，否则为0
    int a = 4; //a = 00000000 00000000 00000000 00000100 
    int b = 6; //b = 00000000 00000000 00000000 00000110

    printf("a = 4 ,b = 6 ,a & b = %d \n" ,a & b);//4

    //2. 位或运算(|) 有1为1，全0则0
    printf("a = 4 ,b = 6 ,a | b = %d \n" ,a | b);//6

    //3. 异或运算(^) 不同为1，相同为0
    printf("a = 4 ,b = 6 ,a ^ b = %d \n" ,a ^ b);//2

    //4. 取反运算(~) 0变1，1变0 按位取反的结果满足： ~x = -(x+1)
    printf("a = 4 ,b = 6 ,~a = %d, ~ b = %d \n" ,~a ,~b);//-5  -7

    //5. 左移运算(<<) 将2进制向左移动指定位数，右边补0
    printf("a = 4，左移2位后为 %d \n" ,a << 2);//a = 00000000 00000000 00000000 00010000 16

    //6. 右移运算(>>) 将2进制向右移动指定位数，正数左边补0 ，负数左边补1
    printf("a = 4，右移1位后为 %d \n" ,a >> 1);//a = 00000000 00000000 00000000 00000010 2
    int c = -2;
    printf("c = -2，右移1位后为 %d \n" ,c >> 1);

    return 0;
}
