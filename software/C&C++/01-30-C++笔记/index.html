    C++

    1. 静态成员变量

    类创建时，如果某个成员加上了static，其存储位置为全局\静态存储区，加了static关键字后，该成员会在类定义的同时被创建，即编译阶段就分配空间，而不是要有对象创建后才分配内存。

    类的静态成员变量必须在类中声明，在类外定义

    静态成员变量不属于某个对象，而是所有该类对象共用的

    和函数中的static变量相同，在编译阶段初始化完成后，不会重新初始化，取值时直接从内存中获取

    1.1 静态成员变量的使用

    #include <iostream>
using namespace std;

class Demo
{
private:
    int a;
public:
    //静态成员变量
    static int b;

    void setA(int n1)
    {
        a = n1;
    }
    int getA(void)
    {
        return a;
    }

    int getB(void)
    {
        return b;
    }
};

//静态成员变量在类中声明，在类外定义，通过 类名::变量名 的方式访问
int Demo::b = 10;

void test_1(void);

int main()
{
    //调用函数实例化类之前就可以访问，说明不属于对象，而是属于类
    cout << "b = " << Demo::b << endl;

    test_1();

    return 0;
}

void test_1(void)
{
    Demo demo0_1;
    Demo demo0_2;

    cout << "demo0_1.b :" << demo0_1.getB() << endl;
    cout << "demo0_2.b :" << demo0_2.getB() << endl;
}

    实例：

    #include <iostream>
using namespace std;

class Demo:
{
public:
    static int counter;
public:
    //每实例化一个对象计数器就+1，每释放一个对象计数器-1
    Demo()//无参构造
    {
        counter ++;
        cout << "counter :" << counter << endl;
    }
    Demo(const Demo& demo)//拷贝构造
    {
        counter ++;
        cout << "counter :" << counter << endl;
    }
    ~Demo()//析构函数
    {
        counter --;
        cout << "counter :" << counter << endl;
    }
};

//类外定义
int Demo::counter = 0;

int main()
{
    //实例化
    Demo demo0_1;
    Demo demo0_2;

    return 0;
}

    2. 静态成员函数

    类中的成员函数加上 static 关键字即为静态成员函数，和静态成员变量相同，在类未实例化之前就分配了内存，可以直接用类名+作用域运算符进行调用。

    静态成员函数主要是用于管理静态成员变量

    静态成员函数只能访问静态成员变量，普通成员变量无法访问

    普通成员函数既可以访问普通成员变量，也可以访问静态成员变量

    例如：

    #include <iostream>
using namespace std;

class Demo
{
private:
    int num1;
    static int a;
public:
    static int getA(void)
    {
        //num1++;//静态成员函数不能访问普通成员变量，只能访问静态成员变量
        return a;
    }
};

//静态成员变量必须在类中声明，类外定义
int Demo::a = 10;

int main()
{
    //未实例化对象之前即可调用
    cout << Demo::getA() << endl;

    return 0;
}

    3. 单例模式

    如果希望类只有一个实例化对象就可以使用单例模式

    单例模式：

    对象在类中实例化，无法由外部进行构造

    单例模式中的对象由静态成员变量存储

    单例模式应该提供一个公用的静态成员函数来访问实例化的对象

    单例模式实现步骤：

    • 私有化构造函数，禁用拷贝构造和重载赋值运算符（=）
    • 使用一个静态成员变量来存储实例化的对象----->存储对象的地址
    • 提供一个公用的静态成员函数来获取对象----->返回地址

    单例模式分为懒汉式和饿汉式实现

    3.1 单例模式---饿汉式实现

    所谓饿汉式实现：不管后面是否用到，先给对象分配一个内存空间

    #include <iostream>
using namespace std;

class Demo
{
private:
    //用一个静态成员变量来存储实例化对象---> 存储地址
    static Demo *instance;
    //私有化构造函数
    Demo(){};
    //禁用拷贝构造
    Demo(const Demo& instance) = delete;
public:
    //提供一个公用的静态成员函数来获取对象地址
    static Demo *getInstance(void)
    {
        return instance;
    }
};

//饿汉式实现先申请实例化类内存
Demo *Demo::instance = new Demo();

void test_1(void);

int main()
{
    test_1();
    /*
    instance_1's address :000001B505699050
    instance_2's address :000001B505699050
    因为只能被实例化一次，因此两次获取的地址都相同
    */

    return 0;
}

void test_1(void)
{
    Demo *instance_1 = Demo::getInstance();
    Demo *instance_2 = Demo::getInstance();

    cout << "instance_1 address :" << instance_1 << endl;
    cout << "instabce_2 address :" << instance_2 << endl;
}

    3.2 单例模式---懒汉式实现

    所谓懒汉式实现：即在需要使用时，才为对象申请内存空间

    #include <iostream>
using namespace std;

class Demo
{
private:
    //用一个静态成员变量来存储对象
    static Demo *instance;
    //私有化构造函数
    Demo(){};
    //禁用拷贝构造
    Demo(const Demo& instance) = delete;
public:
    //提供一个公用的静态成员函数来获取实例化对象的地址
    static Demo *getInstance(void)
    {
        //先判断是否为空地址--->因为一开始未申请内存，这里被访问说明需要用，此时才申请
        if(instance == nullptr)
        {
            instance = new Demo();
        }
        return instance;
    }
};
//静态成员对象在类外定义，此处要将实例化成员对象设置指向空地址
Demo *Demo::instance = nullptr;

void test_1(void);

int main()
{
    test_1();
    /*
    instance_1 :0000012FBD958EE0
    instance_2 :0000012FBD958EE0
    */

    return 0;
}

void test_1(void)
{
    Demo* instance_1 = Demo::getInstance();
    Demo* instance_2 = Demo::getInstance();

    cout << "instance_1 :" << instance_1 << endl;
    cout << "instance_2 :" << instance_2 << endl; 
}

    3.3 单例模式---静态局部变量实现

    因为静态局部变量存储在全局\静态存储区，且第一次创建后，后续不会被重新创建，直到程序结束才被释放，因此可以用静态局部变量来确保只有一个实例化对象

    #include <iostream>
using namespace std;

class Demo
{
private:
    //私有化构造函数
    Demo(){};
    //禁用拷贝构造
    Demo(const Demo& demo) = delete;
public:
    //提供一个公用的静态函数来获取对象地址
    static Demo& getInstance(void)
    {
        //用静态局部变量来存储对象
        static Demo instance;
        return instance;
    }
    /*
    也可以是
    static Demo *getInstance(void)
    {
        static Demo instance;
        return &instance;
    }
    */
};

void test_1(void);

int main()
{
    test_1();

    return 0;
}

void test_1(void)
{
    Demo &instance_1 = Demo::getInstance();
    Demo &instance_2 = Demo::getInstance();

    cout << "instance_1 address :" << &instance_1 << endl;
    cout << "instance_2 address :" << &instance_2 << endl;
}

    从静态函数的函数体中可以看出是在需要用到对象时，才创建的（第一次），因此本质也是懒汉式实现

    4. 类的存储结构

    类本身并不占用存储空间，而是实例化对象后才占空间。

    类中的静态成员变量以及所有函数都不属于类所占的空间，即存储位置和类中普通成员变量不同，计算时不包含在内

    例如：

    #include <iostream>
using namespace std;

class Demo
{
private:
    int a;
    int b;
    float c;
    char d;
    static int n;
public:
    void setA(int n1)
    {   
        a = n1;
    }
    int getA(void)
    {
        return a;   
    }

    void setB(int n1)
    {
        b = n1;
    }
    int getB(void)
    {
        return b;
    }

    void setC(float f1)
    {
        c = f1;
    }
    float getC(void)
    {
        return c;
    }

    void setD(char c1)
    {
        d = c1;
    }
    char getD(void)
    {
        return d;
    }

    Demo()
    {
        a = 10;
        b = 12;
        c = 3.14;
        d = '\0';
    }

    Demo(int n1 ,int n2 ,float n3 ,char n4)
    {
        a = n1;
        b = n2;
        c = n3;
        d = n4;
    }
};

int Demo::n = 10;

int main()
{
    Demo demo0_1;
    Demo demo0_2;

    cout << "sizeof demo0_1 :" << sizeof(demo0_1) << endl;
    cout << "sizeof demo0_2 :" << sizeof(demo0_2) << endl;
    cout << "sizeof Demo :" << sizeof(Demo) << endl;

    /*
    sizeof demo0_1 :16
    sizeof demo0_2 :16
    sizeof Demo :16
    */

    return 0;
}

    为什么是16？

    因为函数和静态成员不计入，静态成员变量和函数都有独立的存储空间，并且被所有实例化对象所共享，因此只需要计算普通成员变量：

    • int a;--->4byte
    • int b;--->4byte
    • float c;--->4byte
    • char d;--->1byte + 3byte （对齐）

    5. this指针

    当实例化类后，通过调用对象中的函数去给函数中的成员变量赋值时，有以下情形：

    #inlcude <iostream>
using namespace std;

class Demo
{
private:
    int a;
    int b;
public:
    void setA(int n1)
    {
        a = n1;
    }
    int getA(void)
    {
        return a;
    }

    void setB(int n1)
    {
        b = n1;
    }
    int getB(void)
    {
        return b;
    }   
};

int main()
{
    Demo demo0_1;
    Demo demo0_2;

    demo0_1.setA(10);
    demo0_2.setA(12);

    cout << "demo0_1.a :" << demo0_1.getA() << endl;//10
    cout << "demo0_2.a :" << demo0_2.getA() << endl;//12

    return 0;
}

    demo0_1和demo0*_*2在调用setA函数时都给各自的a赋值了，但是函数体中直接是

    void setA(int n1)
    {
        a = n1;
    }

    为什么可以知道a是谁的a呢

    加入函数的形参和成员变量名相同呢？

    void setA(int a)
    {
        a = a;//写法错误，编译器会不清楚a所指的是形参还是成员变量
    }

    编译器怎么知道a是指的是形参还是成员变量呢？

    是因为C++对类中的成员函数都封装了一个隐含指针：this指针

    this指针指向被调用的成员函数所属的对象

    上面的例子其实就是：

    void setA(int n1)
    {
        this->a = n1;
    }

    假设是 demo0_1 在调用，则其中 this 指向的就是 demo0_1

    因此当形参和成员变量名相同时，就可以通过this指针说明

    void setA(int a)
    {
        this->a = a;
    }

    注意：

    静态成员函数只能操作静态成员变量，静态成员函数和静态成员变量是所有对象共享的，因此静态成员函数中没有 this 指针

    5.1 this指针的使用---链式编程

    可以将this指针作为普通成员函数的返回值实现链式编程

    #include <iostream>
using namespace std;

class Demo
{
private:
    int a;
    int b;
public:
    Demo()
    {
        a = 0;
        b = 0;
    }
    Demo(int a ,int b)
    {
        this->a = a;
        this->b = b;
    }

    void setA(int a)
    {
        thsi->a = a;
    }
    int getA(void)
    {
        return a;
    }

    //普通成员函数，返回值是this指针，而this指针指向的是对象，因此返回值类型是类对象的别名（也可以用地址）
    Demo& add_num(Demo& demo ,int num)
    {
        int temp = demo.getA();
        temp += num;
        demo.setA(temp);

        return *this;
    }
};

int main()
{
    Demo demo0_1;
    demo0_1.add_num(demo0_1 ,10).add_num(demo0_1 ,20);

    cout << "demo0_1.a :" << demo0_1.getA() << endl;//30

    return 0;
}

    6. const修饰

    6.1 const对类中成员函数的修饰

    类中如果用const修饰成员函数，那么该函数无法对非mutable关键字修饰的成员属性变量进行修改

    #include <iostream>
using namespace std;

class Demo
{
private:
    int a;
    int b;
    mutable int c;//可以在加上const修饰的成员函数中被修改
public:
    void setA(int a)
    {
        this->a = a;
    }
    int getA(void)
    {
        return a;
    }

    void setB(int b)
    {
        this->b = b;
    }
    int getB(void)
    {
        return b;
    }

    Demo() :Demo(0, 0, 0)//这里是使用初始化列表：成员名（包括成员函数）（参数）
    {};

    Demo(int a, int b, int c)
    {
        this->a = a;
        this->b = b;
        this->c = c;
    }

    void print_menber(void)const//加上const修饰后，函数不能修改非mutable修饰的成员
    {
        //a = 100;//若修改则会报 "表达式必须是可修改的左值"
        c = 10;//有mutable修饰，可以修改
        cout << "this is a const func in Demo , a =" << a << "\tb =" << b << "\tc =" << c << endl;
    }

};

int main()
{
    Demo demo1(10 ,20 ,30);

    demo1.print_menber();

    return 0;
}

    6.2 const对类对象的修饰

    使用const修饰的类对象被称为 常对象

    常对象可以访问类中所有成员属性变量，但是无法对非 mutable 修饰的成员属性变量做出修改

    常对象只能访问调用类中用const修饰的成员函数，普通成员函数无法访问调用

    #include <iostream>
using namespace std;

class Demo
{
private:
    int a;
    int b;
    mutable int c;//可以在加上const修饰的成员函数中被修改
public:

    int n1;
    int n2;
    mutable int n3;

    void setA(int a)
    {
        this->a = a;
    }
    int getA(void)
    {
        return a;
    }

    void setB(int b)
    {
        this->b = b;
    }
    int getB(void)
    {
        return b;
    }

    Demo() :Demo(0, 0, 0)//这里是使用初始化列表：成员名（包括成员函数）（参数）
    {};

    Demo(int a, int b, int c)
    {
        this->a = a;
        this->b = b;
        this->c = c;
        n1 = 0;
        n2 = 0;
        n3 = 0;
    }

    void print_menber(void)const//加上const修饰后，函数不能修改非mutable修饰的成员
    {
        //a = 100;//若修改则会报 "表达式必须是可修改的左值"
        c = 10;//有mutable修饰，可以修改
        cout << "this is a const func in Demo , a =" << a << "\tb =" << b << "\tc =" << c << endl;
    }

    void print_n1_3(void)
    {
        cout << "n1 = " << n1 << "n2 = " << n2 << "n3 = " << n3 << endl;
    }

};

int main()
{
    //普通对象
    Demo demo1(10 ,20 ,30);

    demo1.print_menber();

    //常对象
    const Demo demo2(100, 200, 300);
    //可以访问所有成员属性变量，这里只有n1-n3是public权限，a-c是private的
    cout << "demo2 visite demo2.n1:" << demo2.n1 << endl;
    cout << "demo2 visite demo2.n2:" << demo2.n2 << endl;
    cout << "demo2 visite demo2.n3:" << demo2.n3 << endl;

    //非mutable修饰不可修改
    //demo2.n1 = 100;//"表达式必须是可修改的左值"
    demo2.n3 = 13;//mutable修饰，可修改

    //只能调用const修饰的成员函数
    demo2.print_menber();
    //demo2.print_n1_3();//"类型不兼容"

    return 0;
}

    7. 友元

    使用friend关键字可以让类中的所有成员变量和函数被friend修饰的类、类成员函数、全局函数可访问

    friend 关键字只出现在声明处，其他类、类成员函数、全局函数都可声明为友元。友元函数不是类的成员，友元类、友元函数中不可以使用this指针。友元函数可访问对象任意成员属性，包括私有属性

    7.1 普通全局函数作为类的友元

    这里要注意的是全局函数在类中声明时会直接作为函数的全局前向声明，后面无需再单独声明一次

    #include <iostream>
using namespace std;

class Demo
{
//声明友元--->全局函数
    friend void print_demoelement(Demo& demo);
private:
    int age;
    int hight;
    int weight;
public:
    string name;
    string gender;

    Demo() :Demo(0, 0, 0, "nullptr", "nullptr")
    {};

    Demo(int age, int hight, int weight, string name, string gender)
    {
        this->age = age;
        this->hight = hight;
        this->weight = weight;
        this->name = name;
        this->gender = gender;
    }
};

//void print_demoelement(Demo &demo);//友元中会为全局函数做一次全局声明

int main()
{
    Demo demo1(18, 170, 110, "Jack", "Man");
    print_demoelement(demo1);

    return 0;
}

void print_demoelement(Demo& demo)
{
    cout << "age : " << demo.age << endl;
    cout << "hight : " << demo.hight << endl;
    cout << "weight : " << demo.weight << endl;
    cout << "name : " << demo.name << endl;
    cout << "gender : " << demo.gender << endl;
    //可见外部函数可以访问类中的私有成员了
}

    7.2 类的某个成员函数作为类的友元

    注意实例的声明顺序

    #include <iostream>
using namespace std;

class Student;//提前声明

class Teacher
{
public:
    //只声明函数，Student类还没有具体声明，成员属性变量还未知
    void setStudent_Score(Student& student, int yuwen, int shuxue, int yingyu);
};

class Student
{
    friend void Teacher::setStudent_Score(Student& student, int yuwen, int shuxue, int yingyu);
private:
    int yuwen;
    int shuxue;
    int yingyu;
public:
    Student() :Student(0, 0, 0)
    {}
    Student(int yuwen, int shuxue, int yingyu)
    {
        this->yuwen = yuwen;
        this->shuxue = shuxue;
        this->yingyu = yingyu;
    }

    void print_score(void)
    {
        cout << "Chinese : " << yuwen << "\tMath : " << shuxue << "\tEnglish : " << yingyu << endl;
    }
};

int main()
{
    Student the_student;
    Teacher the_teacher;
    the_teacher.setStudent_Score(the_student, 80, 90, 80);

    the_student.print_score();

    return 0;
}

void Teacher::setStudent_Score(Student& student, int yuwen, int shuxue, int yingyu)
{
    student.yuwen = yuwen;
    student.shuxue = shuxue;
    student.yingyu = yingyu;
}

    7.3 整个类为另一个类的友元

    另一个整类被声明为友元时，被声明为友元整类中的所有函数都可以访问声明友元类中的所有成员属性变量（包括const修饰的函数）

    #include <iostream>

#include <iostream>
using namespace std;

//前向声明
class Person;

class Friend1
{
private:
    int age;
    string name;
    int hight;
    int weight;
public:
    Friend1() :Friend1(0, "null", 0, 0)
    { }
    Friend1(int age, string name, int hight, int weight)
    {
        this->age = age;
        this->name = name;
        this->hight = hight;
        this->weight = weight;
    }

    void show_friendmsg(Person person);
    void show_friend2msg(Person person);
};

class Person
{
    friend class Friend1;//类中的所有成员函数（包括const修饰的函数）
private:
    int age;
    string name;
    int hight;
    int weight;
public:
    Person() :Person(0, "null", 0, 0)
    { }
    Person(int age, string name, int hight, int weight)
    {
        this->age = age;
        this->name = name;
        this->hight = hight;
        this->weight = weight;
    }
};

int main()
{
    Person person(18, "Jack", 170, 105);
    Friend1 friend1;

    friend1.show_friendmsg(person);
    friend1.show_friend2msg(person);

    /*
    this is show_friendmsg in Friend1
    person's age :18
    person's name :Jack
    person's hight :170
    person's weight :105

    this is show_friend 2 msg in Friend1
    person's age :18
    person's name :Jack
    person's hight :170
    person's weight :105
    */

    return 0;
}

void Friend1::show_friendmsg(Person person)
{
    cout << "this is show_friendmsg in Friend1" << endl;
    cout << "person's age :" << person.age << endl;
    cout << "person's name :" << person.name << endl;
    cout << "person's hight :" << person.hight << endl;
    cout << "person's weight :" << person.weight << endl;
}

void Friend1::show_friend2msg(Person person)
{
    cout << "this is show_friend 2 msg in Friend1" << endl;
    cout << "person's age :" << person.age << endl;
    cout << "person's name :" << person.name << endl;
    cout << "person's hight :" << person.hight << endl;
    cout << "person's weight :" << person.weight << endl;
}

    注意：

    • 友元不能被继承（父亲的朋友不一定是子女的朋友）
    • 友元是单向的，A是B的友元，B不一定是A的友元（B没有声明A是友元的话）
    • 友元不能传递，A是B的友元，B是C的友元，A不是C的友元
