    C语法基础

    1. 指针进阶

    1.1 多级指针

    以二级指针为例

    type *pointer_name_1 = &data;//一级指针
type **pointer_name_2 = &pointer_name_1

    说明：

    • N级指针指向的是N-1级指针所指向数据的地址
    #include <stdio.h>
#include <stdlib.h>

int main(int argc, char const *argv[])
{
    //普通指针
    char c1 = 'N';

    //一级指针
    char *p1 = &c1;
    printf("p1是指针名，指向的数据类型是char ,指向的数据地址是:%p ，解引用之后是 %c\n" ,p1 ,*p1);
    //p1是指针名，指向的数据类型是char ,指向的数据地址是:000000f340dffa67 ，解引用之后是 N
    //二级指针
    char **p2 = &p1;
    printf("p2是指针名，指向的数据类型是char * ,指向的数据地址是:%p ，解引用之后是 %p\n" ,p2 ,*p2);
    //p2是指针名，指向的数据类型是char * ,指向的数据地址是:000000f340dffa58 ，解引用之后是 000000f340dffa67
    //三级指针
    char ***p3 = &p2;
    printf("p3是指针名，指向的数据类型是char ** ,指向的数据地址是:%p ，解引用之后是 %p\n" ,p3 ,*p3);
    //p3是指针名，指向的数据类型是char ** ,指向的数据地址是:000000f340dffa50 ，解引用之后是 000000f340dffa58

    return 0;
}

    1.2 数组指针&指针数组

    数组指针定义语法：

    type (*pointer_name)[array_size];

    解释：

    一个名称为 pointer_name 的指针，其所指向的是一个有 arraysize 个 type 类型元素的数组整体

    指针数组定义语法：

    type *array_name [array_size];

    解释：

    一个名称为 array_name 的数组，里面有 arraysize 个指向类型为 type 的指针变量

    区分说明：

    *号的优先级低于 [ ] 号，如果要提高其优先级，可以通过添加括号的方式

    因此

    type (*pointer_name)[array_size];

    中的pointer_name必须是指针的名称

    而

    type *array_name [array_size];

    中的array_name必须是数组的名称

    例如：

    #include <stdio.h>

int main(int argc ,const char *argv[])
{
    //数组指针和指针数组
    //数组指针type (*p)[];
    int (*p)[5] = {0};
    printf("p指向的地址是%p \n" ,p);
    //指针数组type *array_name[array_size];
    int *pk[5] = {0};

    /*
    区别在于：
    前者：*的优先级低于[] ,但是加上()后提升了*的优先级,因此这是一个名称为p，指向int [5]数组地址的指针
    后者：一个名称是pk的数组，内部有5个都是int *类型的元素
    */

    return 0;
}

    1.3 void类型指针

    void类型指针即空类型，因此其可以指向任意其他类型

    一般使用场景：

    程序员提前通过malloc()函数开辟好了一块内存，但是由于开辟时还未存储任何类型的数据，此时这块内存就是空类型，但是后续需要使用到这块内存，就只能通过空指针的方式先将这块内存的地址拿到，等到要使用时再将其强制转为使用数据的类型

    注意：

    使用malloc()函数进行内存分配时需要添加该头文件

    #include <stdlib.h>

    手动分配了内存一定要手动释放内存

    free();

    例如：

    #include <stdio.h>
#include <stdlib.h>

int main(int argc, char const *argv[])
{
    void * vp = malloc(20);//开辟了20个byte的内存，但其存储的数据类型没有确定，因此先用一个void类型的指针把地址拿到
    printf("vp= %p \n" ,vp);//vp= 00000236cc8d2420
    //使用该内存需要强转类型
    int *ip = (int *)vp;//转换成要使用的类型
    *ip = 10;
    printf("ip = %p , *ip = %d \n" ,ip ,*ip);//ip = 00000236cc8d2420 , *ip = 10

    //手动释放内存
    free(vp);

    return 0;
}

    分配的内存可以重复使用

    例如：

    #include <stdio.h>
#include <stdlib.h>

int main(int argc, char const *argv[])
{
    void * vp = malloc(20);//开辟了20个byte的内存，但其存储的数据类型没有确定，因此先用一个void类型的指针把地址拿到
    printf("vp= %p \n" ,vp);//vp= 00000236cc8d2420
    //使用该内存需要强转类型
    int *ip = (int *)vp;//转换成要使用的类型
    *ip = 10;
    printf("ip = %p , *ip = %d \n" ,ip ,*ip);//ip = 00000236cc8d2420 , *ip = 10

    float *fp = (float *)vp;
    *fp = 9.9f;
    printf("fp = %p ,*fp = %f\n" ,fp ,*fp);//fp = 000002095afd2420 ,*fp = 9.900000
    printf("ip = %p ,*ip = %d\n" ,ip ,*ip);//ip = 000002095afd2420 ,*ip = 1092511334

    //分配的内存用来存储别的内容了，原来的如果没有被释放就会存到脏数据

    //分配完内存后需要手动释放
    free(vp);

    return 0;
}

    1.4 函数指针数组

    定义语法：

    return_type (*array_name[array_size])(parameter_list) = {func1_name ,func2_name};

    说明：

    一个名称为arrayname的指针数组，内部有arraysize个返回值类型为returntype的函数，这些函数的参数列表都是parameter_list

    注意：

    • 函数指针数组定义时整个数组里每个函数的返回值类型必须都相同

    例子：

    #include <stdio.h>
#include <string.h>

/*
编写一个程序，模拟简单的计算器功能。要求：
- 支持加（`+`）、减（`-`）、乘（``）、除（`/`）四种运算。
- 使用函数指针数组实现功能选择。
- 用户输入两个数字和运算符，程序输出计算结果。
*/

double add_nums(int num1 ,int num2);
double dec_nums(int num1 ,int num2);
double mul_nums(int num1 ,int num2);
double dev_nums(int num1 ,int num2);

int main(int argc, char const *argv[])
{
    int n1 ,n2;
    char c;

    //函数指针数组
    double (*pfunc[])(int ,int) = {add_nums ,dec_nums ,mul_nums ,dev_nums};

    char str[50] = {0};

    while (strcmp(str ,"否") != 0)
    {
        printf("请输入两个整数和功能符号+\\-\\*\\/用空格隔开 \n");

        int input_state = scanf("%d %d %c" ,&n1 ,&n2 ,&c);
        printf("%d %d %c \n" ,n1 ,n2 ,c);
        while (getchar() != '\n');

        if ( input_state != 3)
        {
            printf("输入格式错误！请重新输入 \n");

            continue;
        }

        if (c == '+')
        {
            printf("num1 + num2 = %.2f \n" ,pfunc[0](n1 ,n2));
        }
        else if (c == '-')
        {
            printf("num1 - num2 = %.2f \n" ,pfunc[1](n1 ,n2));
        }
        else if (c == '*')
        {
            printf("num1 * num2 = %.2f \n" ,pfunc[2](n1 ,n2));
        }
        else if (c == '/')
        {
            printf("num1 / num2 = %.2f \n" ,pfunc[3](n1 ,n2));
        }
        else
        {
            printf("输入格式错误！请重新输入 \n");
            while (getchar() != '\n');
        }

        printf("是否还要继续计算？请输入 是 或 否\n");
        scanf("%s" ,&str);
        while (getchar() != '\n');

    }

    printf("结束计算\n");

    return 0;
}

double add_nums(int num1 ,int num2)
{
    double res = (double)num1 + num2;
    return res;
}

double dec_nums(int num1 ,int num2)
{
    double res = (double)num1 - num2;
    return res;
}

double mul_nums(int num1 ,int num2)
{
    double res = (double)num1 * num2;
    return res;
}

double dev_nums(int num1 ,int num2)
{
    double res = (double)num1 / num2;
    return res;
}

    2. 指针常量&常量指针

    2.1 指针常量

    语法：

    type *const pointer_name = &data;

    说明：指针常量是指针本身就是一个常量，而指针存储的是一个地址，即指向的地址不能变，但是地址里面所保存的内容可以变

    例如

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    int a = 10;
    int b = 20;
    int *const pi = &a;
    //但下面这种写法就错了，上面已经确定了pi的地址是a的地址，不能再改成别的地址
    //pi = &b;
    //可以这样
    *pi = 15;//解引用，修改其内容

    return 0;
}

    2.2 常量指针

    语法：

    const type *pointer_name = &data;

    说明：常量指针是指针指向的是一个常量的地址，即该地址所保存的内容不可改变，但是指针所指向的地址可以改变

    例如

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    const int c = 30;
    const int d = 40;
    const int *p3 = &c;
    //可以改变其所指的地址
    p3 = &d;
    //不可以改变其指向的内容
    //*p3 = 45;//解引用后要修改其中的内容

    return 0;
}

    3. 数组进阶

    3.1 数组名相关

    1. 数组名在使用时会转换为数组首元素第一个字节的地址，常量不可更改
    2. 数组名在代码上下文中可以当作是一个指针常量，即其所指向的地址不可改变，但是地址里保存的内容可以改变

    例如

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    //局部变量初始化要定义否则会有脏数据
    int array[6] = {0 ,1 ,2 ,3 ,4 ,5};

    //array = {10 ,11 ,12};
    //这种写法就错误，array在上面定义后就固定指向{0 ,1 ,2 ,3 ,4 ,5}的地址，这种写法相当于是把原来的地址改成{10 ，11 ，12}的地址

    //内容可更改
    printf("array[1] = %d \n" ,*(array+1));//array[1] = 1
    *(array+1) = 15;
    printf("array[1] = %d \n" ,*(array+1));//array[1] = 15

    return 0;
}

    3.2 指针索引（index）的应用

    加法交换法

    例如

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    //加法交换法
    int ar[10];
    for (size_t i = 0; i < 10; i++)//未初始化打印，全是脏数据
    {
        printf("ar[%d] = %d " ,i ,ar[i]);
    }
    printf("\n");
    //写法1
    ar[1] = 777;
    printf("ar[1] = %d \n" ,ar[1]);
    //写法2
    *(ar+1) = 888;
    printf("ar[1] = %d \n" ,ar[1]);
    //写法3
    *(1+ar) = 999;
    printf("ar[1] = %d \n" ,ar[1]);
    //写法4
    1[ar] = 111;
    printf("ar[1] = %d \n" ,ar[1]);

    return 0;
}

    4. 零长数组

    说明：

    即数组长度为0的一个数组

    一般使用场景：

    用于动态分配内存，一般都结合自定义结构体使用，且一般声明在结构体末尾

    注意：

    使用malloc进行内存分配时，最好是使用malloc(sizeof(type)*N) type是零长数组的数据类型，N是要增加的个数

    例如

    #include <stdio.h>
#include <stdlib.h>
#include <string.h>

/*
0长数组：用于动态分配内存，一般都结合自定义结构体使用
且一般声明在结构体末尾
*/

struct Student
{
    int id;
    char gender[10];

    char name[0];
};


int main(int argc, char const *argv[])
{
    //可以通过sizeof输出结构体所占用内存大小，零长数组是0字节
    printf("struct Student所占用内存大小 ： %zu \n" ,sizeof(struct Student));//struct Student所占用内存大小 ： 16

    //为了实现动态分配内存，而直接初始化结构体，是会固定其内存大小，因此只能用结构体+指针变量的形式创建
    struct Student *s1 = (struct Student *)malloc(sizeof(struct Student) + sizeof(char) * 10);//动态分配内存

    //通过指针解引用进行结构体赋值
    //可以使用这种方式
    (*s1).id = 1001;
    //也可以使用这种方式
    s1->id = 1002;
    strcpy(s1->gender ,"男");
    strcpy(s1->name , "张三");

    printf("id = %d ,gender = %s ,name = %s \n" ,s1->id ,s1->gender ,s1->name);//id = 1002 ,gender = 男 ,name = 张三

    //可以使用sizeof查看占用内存大小
    printf("struct Student *s1占用内存大小 = %zu \n" ,sizeof(s1));//是因为s1是一个指针变量，其存储的是一个地址，64位系统中，地址有8字节
    //可以使用strlen查看其字符长度
    printf("struct Student *s1中name项字符长度为 %d ，占用内存大小 = %zu \n" ,strlen(s1->name) ,sizeof(s1->name));//struct Student *s1中name项字符长度为 4 ，占用内存大小 = 0
    //GBK编码模式下，一个中文字符为2字节，而name里面有两个中文
    //而使用sizeof对name项进行计算，输出的结果是0字节，因为sizeof只计算编译期的大小（静态大小，即它计算的还是最上面定义时的name项所占用内存）

    //手动分配内存后一定要手动释放
    free(s1);

    return 0;
}

    5. 多维数组

    5.0 解释

    说明：（以二维数组为例）

    type array_name [N1][N2];

    即第二维是N1个一维数组，其中一维数组中有N2个元素 二维数组中每个元素类型都是相同的

    那么三维数组

    type array_name [N1][N2][N3];

    三维数组则是第三维是N1个二维数组，每个二维数组有N2个一维数组，每个一维数组有N3个元素 以此类推

    例如（以二维数组为例）

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    int array1 [3][2] = {{1 ,2} ,{3 ,4} ,{5 ,6}};

    return 0;
}

    5.1 多维数组的遍历

    嵌套for循环

    #include <stdio.h>

#define LENGTH(array) (sizeof(array)/sizeof(array[0]))

int main(int argc, char const *argv[])
{
    int array1 [3][2] = {{1 ,2} ,{3 ,4} ,{5 ,6}};
    //二维数组的遍历
    for (size_t i = 0; i < LENGTH(array1); i++)
    {
        for (size_t j = 0; j < LENGTH(array1[i]); j++)
        {
            printf("%d " ,array1[i][j]);
        }
        printf("\n");
    }
    printf("\n");

    /*
    1 2 
    3 4
    5 6
    */

    return 0;
}

    5.2 多维数组的初始化

    • 完全初始化方法1

    例如：

    #include <stdio.h>

int main(int argc, char const *argv[])
{
    int array1 [3][2] = {{1 ,2} ,{3 ,4} ,{5 ,6}};

    return 0;
}
    • 完全初始化方法2

    例如：

    #include <stdio.h>

#define LENGTH(array) (sizeof(array)/sizeof(array[0]))

int main(int argc, char const *argv[])
{
    //二维数组初始化
    int array1 [3][2] = {{1 ,2} ,{3 ,4} ,{5 ,6}};
    //或者
    int array2 [3][2] = {10 ,11 ,12 ,13 ,14 ,15};

    return 0;
}
    • 不完全初始化

    例如：

    #include <stdio.h>

#define LENGTH(array) (sizeof(array)/sizeof(array[0]))

int main(int argc, char const *argv[])
{
    //二维数组的不完全初始化
    int array3 [3][2] = {{20} ,{22 ,23} ,{24}};

    return 0;
}
