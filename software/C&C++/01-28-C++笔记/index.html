    C++语法基础

    1. C++和C在内存分配的区别

    1.1 C分配内存

    1.1.1 malloc

    #include <stdio.h>
#include <stdlib.h>

void malloc_use(void);

int main(int argc ,const char *argv[])
{
    malloc_use();
    return 0;
}

void malloc_use(void)
{
    printf("malloc_use\n");
    int* p1 = (int*)malloc(sizeof(int));//分配一个int大小的内存空间，返回的是指向这个空间的指针
    if (p1 == NULL)
    {
        printf("memory distribute fail \n");
        return ;
    }
    *p1 = 5;
    printf("*p1 = %d p1 = %p \n", *p1, p1);

    free(p1);
    p1 = NULL;
}

    1.1.2 calloc

    #include <stdio.h>
#include <stdlib.h>

void calloc_use(void);

int main(int argc ,const char *argv[])
{
    calloc_use();
    return 0;
}

void calloc_use(void)
{
    printf("calloc_use\n");
    char* c1 = (char*)calloc(1, sizeof(char));//分配一个char大小的内存空间并将其先清空，返回指向该空间的指针
    if (c1 == NULL)
    {
        printf("memory distribute fail\n");
        return ;
    }
    *c1 = 'C';
    printf("*c1 = %c c1 = %p \n", *c1, c1);

    free(c1);
    c1 = NULL;
}

    1.1.3 realloc

    #include <stdio.h>
#include <stdlib.h>

void realloc_use(void);

int main(int argc ,const char *argv[])
{
    realloc_use();  

    return 0;
}

void realloc_use(void)
{
    printf("realloc_use\n");
    int* arr1 = (int*)calloc(10, sizeof(int));//10个int类型大小的内存空间
    if (arr1 == NULL)
    {
        printf("memory distribute fail\n");
        return ;
    }
    for (size_t i = 0; i < 10; i++)
    {
        arr1[i] = i + 1;
        printf("%d\t", arr1[i]);
    }
    printf("\n");

    //重新分配内存
    printf("re distribute memory \n");

    int* temp = (int*)realloc(arr1, 20 * sizeof(int));//20个int类型大小的内存空间
    if (temp == NULL)
    {
        printf("memory distribute fail\n");
        free(arr1);
        arr1 = NULL;
        return ;
    }
    arr1 = temp;
    temp = NULL;

    for (size_t i = 10; i < 20; i++)
    {
        arr1[i] = i + 1;
    }

    for (size_t i = 0; i < 20; i++)
    {
        printf("arr1[%d] = %d\n", i, arr1[i]);
    }

    free(arr1);
    arr1 = NULL;
}

    1.2 C++内存分配

    C++分配堆空间主要用以下两个函数：

    • new 申请内存
    • delete 释放内存

    1.2.1 单个变量申请内存

    #include <iostream>
using namespace std;

void signal_distribute(void);

int main()
{
    signal_distribute();    

    return 0;
}

void signal_distribute(void)
{
    cout << "signal_distribute" << endl;
    //单个变量分配内存空间 new 变量类型()
    int* p1 = new int();
    *p1 = 10;
    cout << "*p1 = " << *p1 << " p1 = " << p1 << endl;
    //释放
    delete p1;
}

    使用new申请内存时，若成功则返回指向分配内存的对应类型指针，若失败，默认返回 std::bad_alloc 异常，若程序无捕获则结束

    1.2.2 申请单个变量内存时，可以同时赋初值

    #include <iostream>
using namespace std;

void signal_distribute_withvalue(void);

int main()
{
    signal_distribute_withvalue();

    return 0;
}

void signal_distribute_withvalue(void)
{
    cout << "signal_distribute_withvalue" << endl;
    //单个变量分配内存空间时可以赋初值 new 变量类型 ( 符合类型的初值 )
    char* c1 = new char('f');
    cout << "*c1 = " << *c1 << " c1 = " << (void *)c1 << endl;
    /*
    cout 对 char *类型有特殊处理，会将其当成字符串的指针，打印时会从该处开始打印到 \0 结束
    若想要打印地址，最好是转换为通用类型的指针变量，比如 void *
    */
    //释放
    delete c1;
}

    单个变量释放内存时是 delete 变量指针名

    1.2.3 申请多个连续内存空间

    #include <iostream>
using namespace std;

void multi_distribute(void);

int main()
{
    multi_distribute();

    return 0;
}

void multi_distribute(void)
{
    cout << "multi_distribute" << endl;
    //分配连续的内存空间 new 变量类型[ 变量个数 ]
    int* arr1 = new int[5];
    for (size_t i = 0; i < 5; i++)
    {
        arr1[i] = i + 1;
        cout << "arr1[" << i << "] = " << arr1[i] << endl;
    }
    //释放
    delete []arr1;
}

    1.2.4 申请多个连续内存空间同时赋初值

    #include <iostream>
using namespace std;

void multi_distribute_withvalue(void);

int main()
{
    multi_distribute_withvalue();

    return 0;
}

void multi_distribute_withvalue(void)
{
    cout << "multi_distribute_withvalue" << endl;
    //分配连续的内存空间 new 变量类型[ 变量个数 ] { 第一个数据的初值 ，第二个数据的初值 ... }
    int* arr1 = new int [6] {1, 2, 3, 4, 5, 6};
    for (size_t i = 0; i < 6; i++)
    {
        cout << "arr1[" << i << "] = " << arr1[i] << endl;
    }
    //释放
    delete []arr1;
}

    多个空间释放时 delete [ ]指向该空间的指针名

    2. C++对C的增强

    2.1 全局检测的增强

    C语言运行全局变量中存在同名变量：

    #include <stdio.h>

int n1 = 10;//已赋值，定义

int n1;//未赋值，声明

int main()
{
    printf("n1 = %d \n", n1);//n1 = 10

    return 0;
}

    在C++中不允许该情况，会触发同名冲突

    #include <iostream>
using namespace std;

//C++不允许同一作用域有同名变量
int n1 = 10;
//int n1;//编译不会通过，会报 " int n1 : 重定义 "

int main()
{
    std::cout << "n1= " << n1 << endl;
    return 0;
}

    2.2 C++中所有的变量和函数参数都必须有类型

    C语言中函数参数可以不写类型，表示任意类型

    #include <stdio.h>

int func1(i);

int func2(i);

int func3();//不写参数，可以传任意类型、个数的参数

//不需要参数应该将参数列表设置为void
int func4(void);

//返回值也可以没有类型
g();

int main()
{
    func1(15);

    func2("abc");

    func3(1, 'c', "abc");

    func4();

    printf("g = %d", g());

    return 0;
}

int func1(i)
{
    printf("func1\n");
    printf(" i = %d \n", i);

    return 0;
}

int func2(i)
{
    printf("func2\n");
    printf("%s \n", i);

    return 0;
}

int func3()
{
    printf("func3\n");
    return 0;
}

int func4(void)
{
    printf("func4\n");
    return 0;
}

g()
{
    return 10;
}

    但是这些写法，虽然可以通过编译，但是对应函数不会被执行，因为这些写法其实是老旧的写法，当前编译器判定下其语法不规范，不会被调用执行

    同时以上代码在C++当中无法通过编译，C++当中不允许变量或者参数无类型

    2.3 C++有更严格的类型转换

    C++当中不同类型的变量无法直接赋值，需要使用强制类型转换

    在C语言中：

    #include <stdio.h>
//枚举
typedef enum COLOR
{
    RED ,
    GREEN ,
    YELLOW
}color;

int main()
{
    color mycolor = GREEN;//1

    //直接赋值
    mycolor = 3;

    printf("mycolor = %d \n" ,mycolor);//mycolor = 3

    return 0;
}

    C语言中的枚举类型enum会根据枚举中数据量的大小，由编译器选择被解析为int类型还是unsigned int类型，C语言中可以直接用int类型的数据对其赋值。

    C++当中需要强制类型转换：

    #include <iostream>
using namespace std;

typedef enum COLOR
{
    RED,
    GREEN,
    YELLOW
}color;

int main()
{
    color mycolor = RED;

    //C++中如果要对枚举类型强行赋值需要强制类型转换
    //mycolor = 2;//直接赋值会报错 "不能将 int 类型的值分配到 color 类型的实体"
    mycolor = (color)2;

    cout << "mycolor = " << mycolor << endl;//mycolor = 2

    return 0;
}

    2.4 C++中对struct的增强

    C语言中结构体不能有函数成员，且如果没有加typedef 重命名结构体，实例化结构体时需要加上struct标识

    #include <stdio.h>

//C中结构体不能有函数成员，且快捷使用需要加typedef进行重命名
typedef struct Stu
{
    int age;
    char* name;
}stu;

int main()
{
    //加上了typedef后才能直接使用，否则需要加struct标识
    stu new_student;
    new_student.age = 10;
    new_student.name = "Jack";

    printf("new_student'age = %d\tnew_student'name = %s\n", new_student.age, new_student.name);

    return 0;
}

    C++中结构体可以有函数成员，同时不需要typedef也可以直接实例化结构体

    #include <iostream>
using namespace std;

//C++中允许结构体中有函数成员，并且可以直接使用结构体名实例化结构体，不需要加typedef
struct Stu
{
    int age;
    const char* name;//C++中字符串字面量类型被定义为const char ，C中字面量类型是char
    //函数成员
    void print_stu(Stu stu)
    {
        cout << "student_age :" << stu.age << " student_name :" << stu.name << endl;
    }
};

int main()
{
    //直接实例化结构体
    Stu new_student;
    new_student.age = 10;
    new_student.name = "Whilte";
    //调用结构体成员中的函数
    new_student.print_stu(new_student);

    return 0;
}

    2.5 C++增加了bool类型数据

    C语言中要使用bool类型数据需要添加头文件

    #include <stdbool.h>

    占1byte内存

    #include <stdio.h>
#include <stdbool.h>

int main()
{
    printf("size of bool type is = %zu \n", sizeof(bool));//1type

    return 0;
}

    C++中无需额外添加头文件，已内置bool类型数据，所占内存也是1byte

    #include <iostream>
using namespace std;

int main()
{
    //查看bool类型数据所占内存
    cout << "size of bool type is :" << sizeof(bool) << "byte" << endl;//1bye

    return 0;
}

    2.6 三目运算符增强

    C语言中三目运算返回的结果是一个值（右值），不是一个变量，因此不能再次赋值

    #include <stdio.h>

int main()
{
    //三目运算
    int a = 5, b = 6;
    //(a > b ? a : b) = 10;
    //按理说返回的是 b 将b重新赋值10
    //编译不通过，会报 "表达式必须是可修改的左值" 说明返回的是右值

    return 0;
}

    C++中三目运算返回的是左值，因此可以再操作

    #include <iostream>
using namespace std;

int main()
{
    //三目运算
    int a = 5, b = 6;
    (a > b ? a : b) = 10;
    cout << "a = " << a << " b = " << b << endl;//a = 5   b = 10

    return 0;
}

    2.7 C++对const的增强

    C语言中对于const修饰的变量是默认存储到栈区的，但是若用static修饰的局部const变量和全局const变量是会被存储到全局\静态存储区（只读区），这里要强调的是C语言对于const变量是会直接存储到内存中的（栈区）

    #include <stdio.h>

int main()
{
    const int a = 10;
    //a = 100;//会报错
    //不能对const修饰的变量进行修改

    printf("a = %d \n" ,a);//a = 10

    //但是C语言可以通过指针，将const修饰变量的地址强制转换为普通的指针，通过改变指针所指地址的内容来修改值
    int *p1 = (int *)&a;//这里&a直接对应的应该是const int *p1 = &a;//即常量指针，但是通过强转为普通可变指针，可以让编译器不去检查其const属性
    *p1 = 100;
    printf("a = %d \n" ,a);//a = 100

    return 0;
}

    以上操作在C++中无法重现相同结果，因为C++对于直接用基础数据类型定义的const修饰的变量是放在符号常量表中，不会立刻就分配内存，只有在取地址的时候才会为其分配内存

    #include <iostream>
using namespace std;

int main()
{
    const int a = 10;
    cout << "a = " << a << endl;
    int *p1 = (int *)&a;
    *p1 = 100;
    cout << "a = " << a << " *p1 = " << *p1 << endl;//a = 10  *p1 = 100

    return 0;
}

    所谓符号常量表，编译器在编译期间，就把代码中所有的 a 替换成了 10，因此即便通过指针修改了内存中的值，也不会输出新值

    但是如果const修饰的变量是被另一个变量赋值的，该被赋值变量并不会放到符号常量表中

    #include <iostream>
using namespace std;

int main()
{
    int num0 = 10;
    const int n2 = num0;
    cout << "n2 = " << n2 << endl;
    int* p2 = (int*)&n2;
    *p2 = 200;
    cout << "n2 = " << n2 << " *p2 = " << *p2 << endl;//n2 = 200  *p2 = 200

    return 0;
}

    原因是n2在被定义时就被分配了内存，其存储位置是栈区而不是符号常量表，编译时不会对有n2的地方进行替换

    注意只有基础类型的变量才会有这种情况，像自定义的结构体、类对象是会直接分配栈区内存的

    2.8 C++中优先用const而非#define

    • const需要明确类型，可以被编译器进行类型安全检查，但是 #define 无类型
    • const有明确作用域，而#define 的作用域是从定义到文件结束

    宏常量无类型：

    #include <iostream>
#define PARAM   128

void func(int a);
void func(short a);

int main()
{
    short param = 128;
    func(PARAM);//int
    func(param);//short
    func(128);//int

    return 0;
}

void func(short a)
{
    cout << "short" << endl;
}

void func(int a)
{
    cout << "int" << endl;
}

    宏常量在编译期间会将常量名替换为对应的值，因此会直接调用func(int a)函数，而const有明确的类型，所以指定调用对应函数

    宏常量没有明确作用域：

    #include <iostream>
using namespace std;

#define A   10

void func1(void);

void func2(void);

int main()
{
    func1();
    func2();

    return 0;
}

void func1(void)
{
    const int a = 20;//a仅限于函数内可用
    cout << "a in func1 = " << a << endl;
    cout << "A is #define = " << A << endl;
}

void func2(void)
{
    //cout << "a = " << a << endl;//报错 "未定义标识符 'a' "
    cout << "A is #define = " << A << endl;
}

    若在命名空间中定义宏常量，访问时不需要加上命名空间和作用域运算符，因为宏常量无明确作用域

    #include <iostream>
using namespace std;

namespace A
{
    #define n1  10
}

int main()
{
    //直接访问
    cout << "n1 = " << n1 << endl;//10

    return 0;
}

    2.9 C++中不建议用宏函数而是内联函数

    • 宏函数的参数没有类型，实际使用为了保证语义的正确性还需要对表达式添加括号较为麻烦，内联函数的参数有具体类型可以保证参数完整性和语义
    • 宏函数在预处理阶段就会在代码中进行替换展开，内联函数是在编译阶段才展开
    • 宏函数无作用域，内联函数有作用域
    #include <iostream>
using namespace std;

//宏函数
#define ADD(num1 ,num2) num1+num2  
//其实这里给函数表达式中的参数加上括号提升完整度也可以 ((num1) + (num2))

//内联函数
inline int add_num(int num1, int num2);

int main()
{
    //调用宏函数和内联函数，宏函数如果不对参数加上括号提升完整性可能会破坏语义
    int res1 = ADD(10, 20) * 10;//期望结果是30 * 10 = 300
    int res2 = add_num(10, 20) * 10;//期望结果也是30 * 10 = 300
    cout << "res1 = " << res1 << endl;//10 + 20 * 10 = 210
    cout << "res2 = " << res2 << endl;//30 * 10 = 300

}

inline int add_num(int num1, int num2)
{
    return num1 + num2;
}

    2.10 C++对函数的增强

    2.10.1 函数的默认参数

    在声明时对函数参数赋初值，调用时如果没有指定参数，则使用默认值

    注意：

    • 若函数的声明和定义是分开的，默认参数只能在声明处写，定义处再写，即使和声明处一模一样也会报错
    • 调用函数时若只指定了部分参数，未指定参数则会使用默认值
    • 如果函数的某个参数设置了默认参数，则从其开始往右到参数列表结束，其他参数都要设置默认参数
    • 调用函数时参数是从左到右复制给形参的，因此要确保参数一一对应
    #include <iostream>
using namespace std;

//声明和定义分开，默认参数只能在声明处写
int the_func(int a ,int b = 20 ,int c = 30);//b设置了默认参数，则c也要设置默认参数

int main()
{
    int the_num = 5;
    int resulat = the_func(the_num);//b,c未给出，则使用默认值

    cout << "resulat = " << resulat << endl;//55

    resulat = the_func(the_num, 2, 6);//使用给定的参数计算，参数个数要对应

    cout << "resulat = " << resulat << endl;//13

    return 0;
}

int the_func(int a ,int b ,int c)//定义处不再设置默认参数
{
    return a + b + c;
}

    2.10.2 函数的占位参数

    C++中在声明函数时可以设置占位参数

    占位参数只有参数类型声明，而没有参数名声明。一般情况下，在函数体内部无法使用占位参数。

    #include <iostream>
using namespace std;

int the_func_1(int num1 ,int);//占位参数直接是一个类型名

int the_func_2(int num1 ,int = 10);//占位参数可以设置初始值，函数中不会用到

int main()
{
    //函数调用时占位参数虽然无意义，但依然要传值
    int resu1 = the_func_1(10 ,0);//11
    cout << "resu1 = " << resu1 << endl;

    int resu2 = the_func_2(20);//22
    cout << "resu2 = " << resu2 << endl;

    return 0;
}

int the_func_1(int num1 ,int)
{
    cout << "the_func_1" << endl;
    return num1+1;
}

int the_func_2(int num1 ,int)
{
    cout << "the_func_2" << endl;
    return num1+2;
}

    2.11 重载函数（Overload）

    C++中允许有同名函数：

    同一个作用域中，函数名相同，参数个数不同 参数类型不同 参数顺序不同的函数是可以同时存在并被使用的

    注意：

    • 返回值不能作为重载函数的依据
    • 当使用重载函数时，需要注意占位参数导致函数名的二义性的问题
    #include <iostream>
using namespace std;

int calcu_num(int num1 , double num2);

double calcu_num(double num1 ,double num2);

int calcu_num(int num1 ,int num2 ,int num3);

int calcu_num(double num2 , int num1);

int main()
{
    int the_num1 = 10, the_num2 = 15, the_num3 = 2;

    cout << calcu_num(the_num1, 3.14) << endl;//重载1 13
    cout << calcu_num(5.51, 3.14) << endl;//重载2 1.75478
    cout << calcu_num(the_num1, the_num2, the_num3) << endl;//重载3 50
    cout << calcu_num(3.14 ,the_num2) << endl;//重载4 18

    return 0;
}

int calcu_num(int num1, double num2)
{
    return num1 + num2;
}

double calcu_num(double num1 ,double num2)
{
    return num1 / num2;
}

int calcu_num(int num1 ,int num2 ,int num3)
{
    return (num1 + num2) * num3;
}

int calcu_num(double num2 , int num1)
{
    return num1 + num2;
}

    // 函数的重载和缺省参数一起出现时，要额外的注意二义性
void fun501(int a, int b=10) {
    cout << a << endl;
    cout << b << endl;
}
void fun501(int a) {
    cout << "a=" << a << endl;
}
int main() {
    fun501(10); // 调用时，会报错，因为不知道你到底是要调用只有1个参数的，还是要调用有两个参数的，但是第二个参数是缺省值
    fun501(10);
}
