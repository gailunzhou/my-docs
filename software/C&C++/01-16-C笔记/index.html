    C语法基础

    1. 内存结构

    1.1 虚拟内存布局

    • 栈区：用来存储函数的形参和局部变量，函数进入栈区是以栈帧的形式进入，函数运行结束以后，形参和局部变量都会自动释放，全部由编译器自动执行
    • 堆区：由程序员进行内存的分配，通过调用 malloc() 函数分配内存，使用完成后调用 free() 函数释放内存，如果不手动释放内存，会在程序结束后由系统一起回收，并不是说反正程序结束系统会来回收不用手动去 free() 因为如果长期占用内存而不释放，很容易造成内存溢出

    • 全局、静态存储区：存放加上static关键字的全局变量和静态变量（静态全局变量和静态局部变量）

      • 其中已定义但未初始化的放在 a..bss段 叫动态存值区
      • 已定义且已初始化的放在 b..data段 叫静态存放区
      • 常量区：字符串和加上const关键字修饰的全局变量，该区是只读属性不可更改，局部变量如果加上const关键字还是会存在栈区
      • 代码区：存放源代码的二进制代码

    2. 变量的存储形式

    • auto变量：函数的形参和未加static关键字的局部变量，生命周期：当前函数，存储位置：栈区，作用域：当前函数
    • static变量：已定义未初始化的在a..bss段，定义且初始化的在b..data段，都是全局、静态存储区，加上该关键字后，变量在第一次调用函数初始化后，第二次再次调用不会重新初始化而是直接从全局、静态存储区获得值。生命周期：整个程序，作用域：当前函数
    • register变量：有些变量需要频繁使用，为了减少CPU重复去内存中调用，直接将变量存储在CPU中，从而提高效率，因为变量不在内存中，因此无法通过指针获取其地址，而register变量本质还是auto变量因此生命周期为当前函数

    例如

    #include <stdio.h>

void my_function(int a ,int b);

int main(int argc, char const *argv[])
{
    int n1 ,n2;
    printf("请输入两个正整数，用空格隔开 \n");
    scanf("%d %d" ,&n1 ,&n2);//1 2

    printf("主函数内的局部变量n1 = %d ,地址是 %p \n" ,n1 ,&n1);//主函数内的局部变量n1 = 1 ,地址是 00000051efbffd0c 
    printf("主函数内的局部变量n2 = %d ,地址是 %p \n" ,n2 ,&n2);//主函数内的局部变量n2 = 2 ,地址是 00000051efbffd08

    my_function(n1 ,n2);

    return 0;
}

void my_function(int a ,int b)
{
    printf("形参a：%d，地址是：%p\n" ,a ,&a);//形参a：1，地址是：00000051efbffce0
    printf("形参b：%d，地址是：%p\n" ,b ,&b);//形参b：2，地址是：00000051efbffce8

    int age = a;
    int gender = b;

    printf("局部变量age = %d ,地址是%p \n" ,age ,&age);//局部变量age = 1 ,地址是00000051efbffcbc
    printf("局部变量gender = %d ,地址是%p \n" ,gender ,&gender);//局部变量gender = 2 ,地址是00000051efbffcb8

    static int height = 0;//static局部变量
    printf("请输入函数的静态局部变量height的值（正整数） \n");
    scanf("%d" ,&height);//5
    printf("height = %d ,地址是 %p \n" ,height ,&height);//height = 5 ,地址是 00007ff753036030 

    register int my = 2;
    printf("函数内的register变量my的值为%d ，因为存储位置不在内存中无法通过指针获取地址\n" ,my);

}
    • extern变量：extern变量是全部变量的外部表现形式，是函数外部定义的变量。全局变量的作用域为从定义开始到源文件结束，extern扩展了作用整个程序有效，生命周期：整个程序

    例如以下情况：

    源文件A定义了一个全局变量ARG，主函数所在的源文件MAIN运行程序时需要使用该变量，如果MAIN文件中也定义了一个同样的ARG，

    编译期间并不会报错，而是会在最后的链接期间，编译器会检查到有两个同名的全局变量，调用时不知道该调用哪一个就会报重定义的错误提示，

    但如果源文件A在其头文件中将该变量前加上extern表示外部可引用，

    或者，MAIN文件在程序中加上该变量时前面再加上一个extern表示外部引用就可以解决该问题。

    • extern函数：此外，extern关键字还可以加在函数前面，但是一般情况下函数都是外部可调用的
    • static函数：在函数名前面加上static关键字可以使函数仅在当前文件可用，即使在头文件中声明了，并被别的文件包含了，别的文件也无法调用

    func1源文件：

    #include <stdio.h>

//定义全局变量，全局变量默认有值
int num1 = 10;
int num2;
//由于全局变量是默认可以跨文件调用的，如果想要全局变量仅在当前文件生效需要加static关键字
static int num3 = 13;//加上static关键字后会存储在全局/静态存储区，同时由于已经赋值了，会放在b..data段
static int num4;//存在全局/静态存储区，由于没有赋值，存放在a..bss段

const int num5 = 5;//全局变量前面加上const关键字后，会存储在常量区，该区是只读，不能修改

char *c1 = "cc1";//像字符串这种常量就是存在常量区

void my_func1(void)
{
    printf("func1文件中的全局变量num1 = %d ,地址是 %p \n" ,num1 ,&num1);
    printf("func1文件中的全局变量num2 = %d ,地址是 %p \n" ,num2 ,&num2);

    printf("func1文件中的const全局变量num5 = %d ,地址是 %p \n" ,num5 ,&num5);
    const int num6 = 6;//加上const关键字的局部变量不会存在常量区，还是在栈区
    printf("func1文件中的const局部变量num6 = %d ,地址是 %p \n" ,num6 ,&num6);

    printf("func1文件中的字符串c1 = %s ,地址是 %p \n" ,c1 ,&c1);
}

//函数默认是extern的，如果想要函数仅在当前文件有效也是加static关键字
static void my_func2(void)
{
    printf("func1文件中的static全局变量num3 = %d ,地址是 %p \n" ,num3 ,&num3);
    printf("func1文件中的static全局变量num4 = %d ,地址是 %p \n" ,num4 ,&num4);
}

    global头文件：

    #ifndef __GLOBAL_H
#define __GLOBAL_H


void my_func1(void);
//static void my_func2(void);//不建议声明加上static的函数

#endif

    main.c源文件：

    #include <stdio.h>
#include "..\inc\global.h"

//外部文件中如果要使用其他文件中的全局变量，为了防止重复定义同一个变量，告知编译器该变量已经定义过，需要在变量前面加上extern

//使用extern进行外部声明时，不能同时对变量重新赋值即 extern int num1 = 11; 相当于是在重定义
extern int num1;
extern int num2;
//对于func1中定义的static变量，外部文件无法通过添加extern关键字进行跨文件调用
//static int num3 = 13;//加上static关键字后会存储在全局/静态存储区，同时由于已经赋值了，会放在b..data段
//static int num4;//存在全局/静态存储区，由于没有赋值，存放在a..bss段

extern const int num5;//全局变量前面加上const关键字后，会存储在常量区，该区是只读，不能修改

extern char *c1;//像字符串这种常量就是存在常量区
//不同文件的相同内容字符串是在常量区重新用一个空间存储
char *c2 = "cc1";

int main(int argc, char const *argv[])
{
    my_func1();
    //别的程序无法调用static函数
    //my_func2();

    printf("main文件中的字符串常量 c2 = %s ,地址是 %p \n" ,c2 ,&c2);

    return 0;
}
