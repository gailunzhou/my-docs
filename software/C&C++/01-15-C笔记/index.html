    C语法基础

    1. 位域

    特殊的结构体成员，指定成员的占用位数(2进制)

    例如：

    struct test01
{
    int gender:1;//指定gender只占1位
    int age:4;//4位
    //空域：没有名字
    int :4;//4位
    unsigned cc:4;//一个unsigned int类型(C语言中unsigned 后面如果不写类型默认是 int )的成员 cc是名字 存储空间为4位
};

    和没有指定位数的结构体对比

    #include <stdio.h>

struct test01
{
    int gender:1;//指定gender只占1位
    int age:4;//4位
    //空域：没有名字
    int :4;//4位
    unsigned cc:4;//一个unsigned int类型的成员 cc是名字 存储空间为4位
};

struct test02
{
    int gender;
    int age;
    int unkown;
    unsigned cc;//unsigned如果后面不加类型默认是unsigned int类型
};

struct test03//33位
{
    int gender:1;//指定gender只占1位
    int age:4;//4位
    int c:20;//20位
    //空域：没有名字
    int :4;//4位
    unsigned cc:4;//一个unsigned类型的成员 cc是名字 存储空间为4位
};

int main(int argc, char const *argv[])
{
    //指定位数后，和原来结构体内存计算方式略不同，但还是要为所占内存最大基本成员的倍数
    printf("test01所占内存大小为： %zu \n" ,sizeof(struct test01));//4
    printf("test02所占内存大小为： %zu \n" ,sizeof(struct test02));//16
    printf("test03所占内存大小为： %zu \n" ,sizeof(struct test03));//8

    /*
    test01中按位数计算一共13位，而这里面所占内存最大的基本成员是int类型，因此该结构体最后所占内存应该是int类型4byte的倍数
    而4byte是有32位远大于13位，因此整体最小为4byte
    同理test03中一共33位大于int类型32位，但要保持是int类型的整数倍，因此多的一位会存到下一个4byte空间中一共8byte
    */

    return 0;
}

    说明：

    • 本质还是结构体，只是可以通过该方式指定成员所占位数（限制了其数据范围），结构体所占内存大小的方式还是要按照结构体的规则进行

    由于指定了位数，成员所能表示的数据范围也会因为位数而改变

    例如

    #include <stdio.h>

struct test01
{
    int gender:1;//指定gender只占1位
    int age:4;//4位
    //空域：没有名字
    int :4;//4位
    unsigned cc:4;//一个unsigned类型的成员 cc是名字 存储空间为4位
};

int main(int argc, char const *argv[])
{
    //指定位数后进行赋值需要考虑不同位数的值范围
    //创建一个test01的变量
    struct test01 t1;
    t1.gender = 1;
    printf("t1.gender = %d \n" ,t1.gender);//-1
    /*
    输出结果是-1是因为：设置的只有1位，而如果该位给1，而int是有符号的，此时首位1就变成了符号位-1
    */
    //如果要赋值超过其位数所表达的最大值
    t1.gender = 5;//101
    printf("t1.gender = %d \n" ,t1.gender);//任然只会保留最后一位所以还是-1

    return 0;
}

    2. typedef类型定义

    typedef可以对类型重命名，无法对数据重命名（定义）

    简单使用例：

    #include <stdio.h>

//typedef定义基础数据类型
typedef int new_int;
//typedef定义结构体名
typedef struct Student
{
    new_int age;//使用新定义的类型名
    char gender[10];
    char name[10];
}Stu;

//枚举也可以
typedef enum
{
    Mon,
    Tue,
    Wen,
    Thr,
    Fri,
    Sat,
    Son

}Date;

int main(int argc, char const *argv[])
{
    //使用typedef定义后可快速创建结构体变量
    Stu s1 = {18 ,"男" ,"CC"};
    printf("age = %d\t ,gender = %s\t ,name = %s\t \n" ,s1.age ,s1.gender ,s1.name);

    Date day;
    day = Mon;
    switch (day)
    {
    case Mon:
        printf("Date:%d \n" ,day+1);
        break;

    default:
        break;
    }


    return 0;
}

    3. 宏定义

    • 宏定义的部分一定要用大写字母
    • 宏定义的部分最后只写一行，如果要写成多行，需要在前一行末尾添加 \ 符号来连接下一行
    • 由于宏定义是进行文本替换，编译期间可能会存在逻辑错误，因此最好的写法是为整体加上小括号，并为其中每一个变量都加上小括号

    3.1 无值宏

    #define HONG

    一般使用场景

    头文件

    #ifndef _FILE_H
#define _FILE_H

...

#endif

    条件编译

    #define HONG

#ifdef HONG
...
#else
...
#endif

    3.2 无参宏

    大多是用有意义的英文取替换无意义的数字的场景

    例如

    #define PI  3.1415926

//定义数组长度
#define ARRAY_SIZE  10

    使用例

    #include <stdio.h>

#define PI 3.1415926   

int main(int argc, char const *argv[])
{
    //无参宏一般都定义的是无意义的数字常量
    //例如通过无参宏进行计算
    int r = 10;
    printf("半径为10的圆面积为 %.2f \n" ,PI * r *r);

    return 0;
}

    3.3 有参宏

    使用例

    #include <stdio.h>

#define CMP(a,b) ((a)>(b)?(a):(b))    

int main(int argc, char const *argv[])
{
    //有参宏一定要注意加小括号提升优先级和整体性
    int n1 = 5;
    int n2 = 10;
    printf("%d \n" ,CMP(n1+6 ,CMP(n1 ,n2)));

    return 0;
}

    5. 头文件&文件结构

    • bin文件夹只放用于执行的二进制文件
    • inc文件夹只放头文件(.h文件)
    • src文件夹只放源文件(.c文件)

    头文件可以用于放以下文件：

    • 普通函数声明
    • 宏定义
    • 结构体、联合体模板定义 （声明）
    • 枚举常量列表
    • static 函数和 inline 函数定义（不建议）
    • 其他头文件

    例子：

    头文件：

    #ifndef __SHARE_H
#define __SHARE_H

//其他头文件
#include <string.h>
#include <stdbool.h>

/*
1) 普通函数声明
2) 宏定义
3) 结构体、联合体模板定义 （声明）
4) 枚举常量列表
5) static 函数和 inline 函数定义（不推荐）
6) 其他头文件
*/
//普通函数声明

void print_fail(void);

int add(int num1 ,int num2);

int sub(int num1 ,int num2);

int mul(int num1 ,int num2);

double div(int num1 ,int num2);

//宏定义
#define MAX(a ,b) ((a)>(b)?(a):(b))
#define N   10

//结构体、联合体模板定义 （声明）

{
    int age;
    char gender[10];
    char name[10];

    {

#endif

    6. 使用命令行编译

    编译语法：语法：gcc [选项] 源文件路径/源文件名.c -o 输出可执行文件名

    gcc ./src/hello.c -o hello

    说明： - hello.c：C源文件（存放main()函数的文件） - -o hello：指定编译后可执行的文件名hello（不加-o的话，默认生成a.out（Linux/macOS）或a.exe（Windows））

    编译多个源文件：

    gcc ./src/hello.c ./src/main.c -o hello

    如果头文件存和源文件存放在不同位置，则需要指定头文件路径

    编译语法：语法：gcc [选项] 源文件路径/源文件名.c -I头文件路径 -o 输出可执行文件名

    gcc ./src/hello.c ./src/main.c -I./inc -o hello.exe

    说明： - 如果不加-I，则会默认在当前路径下(src目录下)去找源文件中包含的头文件(例如这里加上为hello.h)，如果找不到会报找不到文件的错误，因此该情况下需要指定头文件路径 - 可以指定输出位置：-o ./bin/hello.exe（一般文件夹bin目录就是用来存放执行文件的）

    课外

    1. 随机数

    生成一个随机数需要包含两个头文件

    #include <stdlib.h>
#include <time.h>

    生成随机数的步骤：

    • 初始化随机数生成器
    • 生成随机数

    1.1 初始化随机数生成器

    初始化随机数生成器使用的函数是

    srand(unsigned int seed);

    其中的seed是随机数种子，默认值是 1

    如果不对seed进行设置，那么每一次生成的随机数列都是相同的，即：

    第一次调用生成随机数：

    第一个数是43；

    第二个数是57；

    ...

    第二次调用生成随机数：

    第一个数是43；

    第二个数是57；

    ...

    其实没有达到随机的目的

    因此一般的做法是：

    通过获取系统当前的时间戳（秒数），将这个时间戳作为seed传入，以实现每一次调用都不同

    根据函数的定义，参数类型必须为 unsigned int 类型，因此可能需要进行强制类型转换

    srand((unsigned int) time(NULL));
//或者
srand((unsigned) time(NULL));//C语言中 unsigned 如果后面不指定的话默认是 unsigned int
    1.1.1 如何获得当前系统时间戳

    获取当前系统时间戳的函数是：

    time(time_t *seconds);

    参数里的 time_t* seconds 是一个指向 time_t 类型名称为 seconds 的指针，用来存储获取的时间值

    所以也可以直接填入 NULL 即 仅获取时间戳不存储时间值

    1.2 生成随机数

    使用 srand() 函数初始化完成后就可以开始生成随机数了

    生成随机数的函数是：

    int rand(void);

    该函数会生成从1到 RAND_MAX 的整数

    这里的 RAND_MAX 是一个常量，它的默认值在不同的实现中会有所不同，但是值至少是 32767

    因此如果需要对生成的随机数限制范围可以通过取余的方式

    例如：

    //获得0到99的随机数
int num = rand() % 100;
//通过 +1 获得1到100的随机数
num = rand() % 100 + 1;
//获得0到49的随机数
num = rand() % 50;

    至此，获取一个随机数的完整流程参考以下代码：

    #include <stdio.h>
#include <stdlib.h>
#include <time.h>

int main(int argc ,const char *argv[])
{
    //1.创建一个time_t类型的数据(可选)
    time_t t;   
    //2.初始化随机数生成器
    srand((unsigned int) time(&t));
    //或者 srand((unsigned)time(NULL));
    //3.获取随机数
    int num = rand();
    //4.打印获取的随机数
    printf("获取的随机数是 %d \n" ,num);

    return 0;
}
