# C语法基础

## 1. 函数基础

函数定义语法：

```c
return_type function_name (parameter_list)
{
	function_body

	return 返回值;
}
```

**说明：**

*   return\_type是函数返回值的数据类型，有些函数**需要执行一定操作而不返回值**，此时可以没有 return 返回值；同时函数的returnt\_ype 是 **void**
*   **函数名不能重复**
*   函数的参数可以为空，即return\_type function\_name(void)
*   函数需要**先声明，再定义**
*   函数要**高内聚，低耦合---所使用的变量尽量都为函数内部的局部变量**

函数的基础创建以及使用：

```c
#include <stdio.h>

//声明函数
int add_nums(int num1 ,int num2);
int sub_nums(int num1 ,int num2);
int mul_nums(int num1 ,int num2);
double dev_nums(int num1 ,int num2);

int main(int argc, char const *argv[])
{
    int n1 ,n2;
    char c;

    printf("请输入两个整数和功能符号+\\-\\*\\/用空格隔开 \n");
    scanf("%d %d %c" ,&n1 ,&n2 ,&c);

    switch (c)
    {
    case '+':
        printf("num1 + num2 = %d \n" ,add_nums(n1 ,n2));//使用函数
        break;
    case '-':
        printf("num1 - num2 = %d \n" ,sub_nums(n1 ,n2));//使用函数
        break;
    case '*':
        printf("num1 * num2 = %d \n" ,mul_nums(n1 ,n2));//使用函数
        break;
    case '/':
        printf("num1 / num2 = %.2f \n" ,dev_nums(n1 ,n2));//使用函数
        break;
    
    default:
        break;
    }

    return 0;
}

//定义函数
int add_nums(int num1 ,int num2)
{
    int res = num1 + num2;
    return res;
}

int sub_nums(int num1 ,int num2)
{
    int res = num1 - num2;
    return res;
}

int mul_nums(int num1 ,int num2)
{
    int res = num1 * num2;
    return res;
}

double dev_nums(int num1 ,int num2)
{
    double res = (double)num1 / (double)num2;
    return res;
}

```

## 2. 实参和形参

实参：**函数被调用时给函数传入的参数**

形参：**创建函数时在参数列表里的参数**

函数调用的本质：**值传递**-----将传入的参数复制一份给到函数内部的形参，由于**实参和形参存储的位置不同**，因此**形参和实参是互不影响**

传普通变量：

```c
#include <stdio.h>

int add_nums(int num1 ,int num2);

int main(int argc, char const *argv[])
{
    //传普通变量
    int n1 ,n2;
    printf("输入两个整数，用空格隔开 \n");
    scanf("%d %d" ,&n1 ,&n2);

    printf("传入之前的n1 = %d ,n2 = %d \n" ,n1 ,n2);//传入之前的n1 = 2 ,n2 = 5 
    add_nums(n1 ,n2);
    printf("传入之后的n1 = %d ,n2 = %d \n" ,n1 ,n2);//传入之后的n1 = 2 ,n2 = 5

    return 0;
}

int add_nums(int num1 ,int num2)
{
    int res = num1 + num2;
    printf("num1 = %d ,num2 = %d \n" ,num1 ,num2);//num1 = 2 ,num2 = 5

    return res;
}

```

传指针变量：

```c
#include <stdio.h>

int aDD_NUMS(int *p1 ,int *p2);

int main(int argc, char const *argv[])
{
    int n1 ,n2;
    printf("输入两个整数，用空格隔开 \n");
    scanf("%d %d" ,&n1 ,&n2);

    //传指针变量：传的是地址，若函数内解引用并修改了内容，原地址不会改变，但内容会改变
    int *np1 = &n1;
    int *np2 = &n2;

    printf("传入之前的np1地址:%p ,内容是:%d,np2地址:%p ,内容是:%d \n" ,np1 ,*np1 ,np2 ,*np2);//传入之前的np1地址:0000003a91bffa6c ,内容是:2,np2地址:0000003a91bffa68 ,内容是:5
    aDD_NUMS(np1 ,np2);
    printf("传入之后的np1地址:%p ,内容是:%d,np2地址:%p ,内容是:%d \n" ,np1 ,*np1 ,np2 ,*np2);//传入之后的np1地址:0000003a91bffa6c ,内容是:10,np2地址:0000003a91bffa68 ,内容是:20

    return 0;
}

int aDD_NUMS(int *p1 ,int *p2)
{  
    printf("p1的地址是：%p ,内容是:%d ,p2的地址是：%p ,内容是:%d \n" ,p1 ,*p1 ,p2 ,*p2);//p1的地址是：0000003a91bffa6c ,内容是:2 ,p2的地址是：0000003a91bffa68 ,内容是:5
    *p1 = 10;
    *p2 = 20;
    int res = *p1 + *p2;

    return res;
}
```

## 3. 关于strcpy()和指针

### 3.1 字符串和字符数组的区别

区别：**存储地方不同，字符串是存储在内存的常量区，字符数组是存储在栈区**

```c
#include <stdio.h>
#include <string.h>

int main(int argc, char const *argv[])
{
    //字符串和字符数组的区别
    //字符串
    char str[] = "hello";
    char str1[] = {'h' ,'e' ,'l' ,'l' ,'o' ,'\0'};

    //都使用puts函数输出，结果都一样，但str后面的hello是存在常量区，str1后面的内容是存在栈区
    puts(str);
    puts(str1);

    return 0;
}

```

### 3.2 关于strcpy()

```c
#include <stdio.h>
#include <string.h>

int main(int argc, char const *argv[])
{

    //关于strcpy
    //puts(strcpy("777" ,"888"));//该代码不运行，原因是字符串都是存在常量区，复制操作相当于是要改动字符串，而常量不允许改动
    //但是这种可以运行：
    char str2[] = "777";
    puts(strcpy(str2 ,"888"));
    //原因是str2不是存在常量区，而其中的"777"是从常量区复制过来然后存在栈区的，str2是一个指向str2数组的指针，保存的是str2的地址，strcpy更改的是栈区的内容，也就是将str2指向地址的内容更改，因此可以运行

    return 0;
}
```

### 3.3 关于“指针就是指针”

```c
#include <stdio.h>
#include <string.h>

int main(int argc, char const *argv[])
{
    char str[] = "hello";

    //关于指针就是指针
    char *pstr = "hello";
    //pstr指向的是在常量区"hello"字符串的基地址 ，str指向的是从常量区的"hello"拷贝到栈区中的"hello"的地址
    printf("*pstr所指向的地址是%p \n" ,pstr);//*pstr所指向的地址是00007ff7c26ea000 
    printf("str所指向的地址是%p \n" ,str);//str所指向的地址是00000028517ffc02 

    return 0;
}
```

## 4. 结构体

### 4.1 创建结构体

```c
struct 结构体名 
{
    成员1;
    成员2;
    成员3;
    成员...;
};

可保存多个类型的成员
```

创建和使用结构体：

```c
#include <stdio.h>
#include <string.h>

//创建结构体
struct Student
{
    char name[10];
    int id;
    char gender[10];
    double height;
    double weight;
};

int main(int argc, char const *argv[])
{
    //使用结构体
    struct Student student1 = {"张三" ,9527 ,"男" ,175.20 ,135.25};
    //访问结构体内的成员
    printf("student1的名字: %s \n" ,student1.name);
    printf("student1的id : %d \n" ,student1.id);
    printf("student1的性别 : %s \n" ,student1.gender);
    printf("student1的身高 : %.2f \n" ,student1.height);
    printf("student1的体重 : %.2f \n" ,student1.weight);

    printf("\n");

    return 0;
}
```

通过创建函数，传入结构体作为参数进行快速访问（减少重复代码）:

通过指针访问、修改结构体内容：

**注意：字符串类型的只能通过strcpy()进行修改**

```c
#include <stdio.h>
#include <string.h>

//创建结构体
struct Student
{
    char name[10];
    int id;
    char gender[10];
    double height;
    double weight;
};

//通过创建函数参数为自定义的结构体，实现快速访问
void show_student(struct Student student);

int main(int argc, char const *argv[])
{
    //使用结构体
    struct Student student1 = {"张三" ,9527 ,"男" ,175.20 ,135.25};
    //访问结构体内的成员
    printf("student1的名字: %s \n" ,student1.name);
    printf("student1的id : %d \n" ,student1.id);
    printf("student1的性别 : %s \n" ,student1.gender);
    printf("student1的身高 : %.2f \n" ,student1.height);
    printf("student1的体重 : %.2f \n" ,student1.weight);

    printf("\n");


    //可以使用指针

    struct Student student3 = {"李四" ,9537 ,"男" ,177.88 ,145.22};
    
    struct Student *s3p = &student3;

    show_student(*s3p);

    printf("\n");

    //通过指针修改结构体成员内容
    (*s3p).id = 20;
    //或
    s3p -> height = 188.82;
    
    show_student((*s3p));
    /*
    字符串类型的只能通过strcpy()进行修改，
    原因是最开始赋值时已经在常量区开辟了一块空间存储所赋值的字符串
    比如：
    struct Student student1 = {"张三" ,9527 ,"男" ,175.20 ,135.25};
    中，假设现在要更改姓名
    student1.name = "李四";
    ↑ 这种写法是错误的，原因：
    最开始创建结构体的时候：char name[10];
    当赋值了以后，属于student1的name字符串（本质就是一个字符组成的数组）其指向的地址已经确定("张三"的地址)，
    赋值时是先在常量区开辟一个地方存储"张三"再把其复制，粘贴到栈区的一个地方，student.name就是指的该地方的地址
    如果使用=进行重新赋值，就相当于要把原来存储"张三"的地址给赋值为存储"李四"的地址，但地址是一个常量，不允许被赋值
    */
    //更改字符串的内容
    printf("\n");

    strcpy(student1.name ,"李四");
    show_student(student1);

    printf("\n");
}

void show_student(struct Student student)
{
    printf("student的名字: %s \n" ,student.name);
    printf("student的id : %d \n" ,student.id);
    printf("student的性别 : %s \n" ,student.gender);
    printf("student的身高 : %.2f \n" ,student.height);
    printf("student的体重 : %.2f \n" ,student.weight);
}

```

**字符串类型的只能通过strcpy()进行修改，是因为如果最开始结构体定义的时候就是数组类型的话，那么第一次赋值了以后，这个量直接就是一个地址，地址是常量，后续无法更改**

但是如果最开始定义结构体的时候是定义的指针类型的数据，那么后面就可以进行直接赋值，**因为如果是指针类型的数据，其第一次初始化以后，相当于是只是让指针指向你给的字符串的地址，但是后续还是可以更改，因为更改只是让这个指针指向别的字符串的地址**

其本质就是：**如果定义的是数组，其存储的是地址，如果定义的是指针，其就是一个指针变量**

## 5. 关于入栈和出栈

关键理解：

*   **esp:始终指向栈顶**
*   **ebp:始终指向一个函数的栈底**
*   **栈的结构是高地址在上，低地址在下，“倒扣的水杯”**

函数A调用函数B时：

1.  一开始ebp指向函数A的栈底，esp指向函数A的栈顶
2.  esp先向下移动一个空间，存储指向函数A栈底的ebp内容，称之为 旧ebp ，然后esp再向下移动一个空间，该位置就是函数B的ebp位置，然后esp继续向下移动，为函数B的局部变量腾出空间，esp就到达新的栈顶
3.  当函数B调用结束时，esp会回到旧ebp的位置，把其中内容取出来交给ebp，然后ebp就会回到之前函数A栈底的位置，然后esp可能会继续往上移动，清理给调用函数B的参数以及返回地址
