# 内存分配

## 1. 操作内存空间

### 1.1 申请内存

程序员对内存操作都是操作堆空间，C语言中要对内存进行操作必须包含该头文件:

```c
#include <stdlib.h>
```

#### malloc()函数

函数定义：

```c
void *malloc(size_t size);
```

*   根据定义可知**函数参数是指要申请的字节数，函数返回的是一个void类型的指针，指向已分配大小的内存。如果请求失败，则返回 NULL**
*   但是**该函数并不会对所申请的内存进行清理，即申请到的内存空间可能会有脏数据**
*   size是指字节数，实际使用时**建议使用 N\*sizeof( type ) 的方式获取**

简单使用例:

```c
#include <stdio.h>
#include <stdlib.h>

int main(int argc ,const char *argv[])
{
	//调用malloc函数申请4byte大小内存
	void *vp = malloc(4);//用一个void类型的指针存储返回值
	
	//一个int类型数据就是4byte，这里把地址再赋值给另一个int类型的指针
	int *ip = (int *)vp;//因为vp是void类型的指针，这里强制类型转为int类型指针才能赋值给ip

	*ip = 5;//解引用赋值5
	printf("ip的地址是 %p ,vp的地址是 %p ,ip里的内容是 %d \n" ,ip ,vp ,*ip);

	//申请之后一定要释放内存
	free(vp);
	vp = NULL;//避免野指针
	ip = NULL;

	return 0;
}

```

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int main(int argc ,const char *argv[])
{
	//调用malloc函数申请15byte大小内存，推荐使用 N*sizeof(数据类型) 的方式
	char *str = (char *)malloc(15*sizeof(char));//可以直接强制类型转换赋值给目标指针

	strcpy(str ,"runoob");//字符串名就是地址，字符串重新赋值只能通过strcpy复制粘贴
	printf("str的地址是 %p ,str里的内容是 %s \n" ,str ,str);

	//申请之后一定要释放内存
	free(str);
	str = NULL;

	return 0;
}
```

当成一个数组来用:

```c
#include <stdio.h>
#include <stdlib.h>

int main(int argc ,const char *argv[])
{
	//申请一个5*sizeof(int)大小的内存，当成一个 int [5] 的数组
	int *ip = (int *)malloc(5 * sizeof(int));
	//给数组赋值
	for(size_t i = 0; i < 5 ;i++)
	{
		ip[i] = i;//数组名就是地址
	}
	//遍历数组
	for(size_t i = 0; i < 5 ;i++)
	{
		printf("ip[%d]的内容是 %d \n" ,i ,ip[i]);
	}

	//释放内存
	free(ip);
	ip = NULL;

	return 0;
}
```

#### calloc()函数

上面malloc()函数中提到，malloc不会对获取的内存进行清空，因此里面可能存在脏数据

使用**calloc函数可以实现申请并将获取到的内存清空**

定义：

```c
void *calloc(size_t nitems, size_t size);
```

*   **nitems 是指要被分配的元素个数**
*   **size 是指每个元素的数据大小**

简单使用例：

```c
#include <stdio.h>
#include <stdlib.h>

int main(int argc ,const char *argv[])
{
	int n = 0;
	printf("请输入要输入的元素个数 n（正整数）：");
	scanf("%d" ,&n);

	int *ip = (int *)calloc(n ,sizeof(int));//申请并清空n个int类型数据大小(4byte)的内存

	printf("请输入n个正整数，每个以回车结束\n");
	for(size_t i = 0; i < n; i++)
	{
		scanf("%d" ,&ip[i]);
	}

	//遍历数组
	for(size_t i = 0; i < n; i++)
	{
		printf("%d\n" ,ip[i]);
	}

	//释放内存
	free(ip);
	ip = NULL;	

	return 0;
}

```

#### realloc()函数

上面的malloc()和calloc()函数都是分配了固定的内存空间，但是如果现在需要放更多的内容，那就需要将分配的内存扩大

使用**realloc()函数可以对指向通过malloc()或者calloc()分配内存的指针进行重新分配**

定义：

```c
void *realloc(void *ptr, size_t size);
```

*   **ptr 是一个指向要重新分配内存的指针，该指针原来指向通过malloc()、calloc()或者realloc()函数分配的内存块，如果是一个空指针，则其会指向realloc()此时申请到的内存块。**
*   **size 是要重新分配的内存大小，如果填0，则会把 ptr 所指的内存释放，并且 ptr 会指向NULL，避免了野指针。**

简单使用例：

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int main(int argc ,const char *argv[])
{
	char *str = (char *)calloc(15 ,sizeof(char));
	strcpy(str ,"runboo");
	printf("str的地址是 %p ,内容是 %s \n" ,str ,str);

	//重新分配内存
	char *temp = (char *)realloc(str ,25 * sizeof(char));
	str = temp;
	temp = NULL;
	strcat(str ,".com");
	printf("str的地址是 %p ,内容是 %s \n" ,str ,str);

	free(str);
	str = NULL;

	return 0;
}
```

说明：

使用realloc()进行重新分配内存时，以上面为例，**如果直接覆盖原来的**：

```c
str = (char *)realloc(str ,25 * sizeof(char));
```

**如果realloc分配失败，即没有足够大小的内存时，会返回NULL，此时 str 指向NULL，而str原来指向的空间则无法被释放，因此造成内存泄漏，因此为了避免这种情况，应该使用一个临时的指针先存储，然后再赋值给原有的指针，并让临时指针指向NULL避免野指针**

```c
//第一次分配内存
char *str = (char *)calloc(15 ,sizeof(char));

//重新分配内存
char *temp = (char *)realloc(str ,25 * sizeof(char));
str = temp;
temp = NULL;
```

因此可以实现动态数组：

```c
#include <stdio.h>
#include <stdlib.h>

int main(int argc ,const char *argv[])
{
	//分配一个长度为5的int型数组
	int *ip = (int *)calloc(5 ,sizeof(int));
	if(ip == NULL)
	{
		printf("内存分配失败，程序结束\n");
		return 1;
	}
	//存入数据
	for(size_t i = 0 ; i < 5 ; i++)
	{
		ip[i] = i;
	}
	//遍历
	for(size_t i = 0 ; i < 5 ; i++)
	{
		printf("ip[%d] = %d " ,i ,ip[i]);
	}
	printf("\n");

	//重新分配内存，原有基础上再增加5长度
	int *temp = (int *)realloc(ip ,10 * sizeof(int));
	if(temp == 0)
	{
		printf("内存重分配失败，程序结束\n");
		//释放原来的内存
		free(ip);
		ip = NULL;
		return 1;
	}
	ip = temp;
	temp = NULL;

	//存入新数据
	for(size_t i = 5 ; i < 10 ; i++)
	{
		ip[i] = i;
	}
	//遍历
	for(size_t i = 0 ; i < 10 ; i++)
	{
		printf("ip[%d] = %d " ,i ,ip[i]);
	}

	//释放内存
	free(ip);
	ip = NULL;

	return 0;
}
```

新增学员部分：

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

//定义学员结构体（全局）应该放在头文件的
typedef struct
{
	int id;//学员编号
	char name[32];//学员姓名
	double score[3];//三科成绩数组
	double total;//总分
}student;

//学员人数（全局）
int real_stu_num;

//学员数组（全局）
student *sp;

//新增学员 尾部追加
void add_student(void)
{
	int id = 0;//学员编号
	char name[32];//学员姓名
	int yuwen,shuxue,yingyu;//语文，数学，英语成绩

	if(real_stu_num == 0)//如果是第一个
	{
		//申请一个student大小的内存
		sp = (student *)calloc(1 ,sizeof(student));
		if(sp == NULL)
		{
			printf("申请内存失败\n");
			return;
		}
		
		printf("请输入新学员编号，第一个学员编号为1001，请按该格式填写：");
		scanf("%d" ,&id);

		//填入编号
		sp[0].id = id;
	}
	else
	{
		//如果前面已有学员
		//重新分配内存
		student *temp = (student *)realloc(sp ,(real_stu_num + 1) * sizeof(student));
		if(temp == NULL)
		{
			printf("申请内存失败\n");
			free(sp);
			sp = NULL;
			return;
		}
		sp = temp;
		temp = NULL;

		printf("请输入新学员编号，第一个学员编号为1001，请按该格式填写：");
		scanf("%d" ,&id);

		int id_pass_flag = 0;//id可行标志位

		while(!id_pass_flag)
		{
			id_pass_flag = 1;//假设id可行
			for(size_t i = 0; i < real_stu_num ;i++)
			{
				if(id == sp[i].id)
				{
					id_pass_flag = 0;
					printf("该编号已有学员使用，请重新输入：");
					scanf("%d" ,&id);
					break;//找到了就直接停止for循环后面的项
				}
			}		

		}
		//填入编号
		sp[real_stu_num].id = id;
	}
		printf("请输入新学员姓名:");
		scanf("%s" ,name);
		//字符串重新赋值只能复制粘贴
		strcpy(sp[real_stu_num].name ,name);

		printf("请输入三门课程成绩(每一门之间用一个空格隔开)\t语文\t数学\t英语\n");
		scanf("%d %d %d" ,&yuwen ,&shuxue ,&yingyu);
		sp[real_stu_num].score[0] = (double)yuwen;
		sp[real_stu_num].score[1] = (double)shuxue;
		sp[real_stu_num].score[2] = (double)yingyu;

		//总分
		sp[real_stu_num].total = sp[real_stu_num].score[0] + sp[real_stu_num].score[1] + sp[real_stu_num].score[2];
		
		real_stu_num ++;//学员人数+1
		//添加完成后输出提示
		printf("添加成功！\n");
}
```

删除学员：

```c
#include <stdio.h>
#include <stdlib.h>
#include <>

//外部引用学员人数（全局）
extern int real_stu_num;

//外部引用学员数组（全局）
extern student *sp;

extern int search_id(void);//外部引用函数

void delete_student(void)
{
	if(real_stu_num == 0)
	{
		printf("暂无学员，请重试\n");
	}
	else
	{
		int index;
		index = search_id();
		if(index == -1)
		{
			printf("无该编号学员\n");
		}
		else
		{
			for(size_t i = index ; i < real_stu_num - 1 ; i++)//从选择的那位开始到最后一位
			{
				sp[i] = sp[i+1];//用后一项的数据覆盖前一项
			}

			real_stu_num --;//学生人数减1
			
			if(real_stu_num > 0)//如果还有学生就重新分配内存（直接释放掉多的内存）
			{
				student *temp = (student *)realloc(sp ,real_stu_num * sizeof(student));
				if(temp == NULL)
				{
					printf("重新分配内存失败\n");
					free(sp);
					sp = NULL;
					return;
				}
				sp = temp;
				temp = NULL;
			}
			else//如果没有学生了
			{
				free(sp);//释放内存
				sp = NULL;
			}

			printf("删除成功\n");
		}
	}
}
```
