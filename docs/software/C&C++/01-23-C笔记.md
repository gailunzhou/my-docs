# C算法基础

## 1. 递归算法

**递归是函数自己在函数体内调用自己，因此只适用于逻辑一样的，例如公式计算**，同时在递归过程中，在内存当中栈帧是一直被加入到栈里，CPU并不知道要调用多少层，如果函数体中存在停止条件，则会运行到停止再开始一层一层回收，因此**不适用于那种逻辑较为复杂的，因为很容易造成栈溢出**。

### 1.1 实例

下面以斐波那契数列计算为例

头文件：

```c
#ifndef __FIBONACCI_H
#define __FIBONACCI_H

//函数声明
int fibonacci(int num);

#endif
```

源文件：

```c
#include <stdio.h>
#include <stdlib.h>
#include "..\inc\fibonacci.h"

//参数：斐波那契数列数列的n

/*
斐波那契数列：
0，1，1，2，3，5，8...
从第2项开始，f(n) = f(n-1) + f(n-2) (n >= 2)
*/

int fibonacci(int num)
{
	if(num < 0)
	{
		printf("输入错误，函数退出\n");
		exit(1);
	}
	else if(num == 1 || num == 0)
	{
		return num;
	}
	else
	{
		return fibonacci(num - 1) + fibonacci(num - 2);
	}
}

```

主函数文件：

```c
#include <stdio.h>
#include "..\inc\fibonacci.h"

int main(int argc ,const *char argv[])
{
	int n;
	printf("请输入要计算斐波那契数列的n:");
	scanf("%d" ,&n);

	//从0开始列出
	for(size_t i = 0; i < n ;i++)
	{
		printf("%d " ,fibonacci(i));
	}

	return 0;
}
```

## 2. 冒泡排序

以数组为例

假设现在有一个长度为100的int类型数组array\[100]，里面的数大小随机，现在要对其进行从小到大排序（左小右大）

使用冒泡排序的方法：

*   从第一个数开始array\[0]，记为max，和其后一个数array\[1]比大小，如果array\[0] > array\[1]则互换位置，反之则不动
*   然后再从第二个数array\[1] (如果换过位置，现在里面是array\[0]的数) ，和array\[2]比大小，如果array\[1] > array\[2] ，则互换位置，反之则不动
*   然后从array\[2]开始(如果换过位置，现在里面是array\[1]的数），和array\[3]比大小，如果array\[2] > array\[3] 则互换位置，反之则不动
*   重复上述过程，第一轮就可以得出当前最大数，并放在最右边
*   开始第二轮，重复上述过程，就可以得出数组中第二大的数，并放在从右向左的第二位
*   一共要重复 n - 1轮 （n为数组长度，例如这里的array\[100] ，那就只用比较99轮）就可以将数从小到大排好序，为什么是 n - 1轮？因为假设数组长度为100 ，当比较到99轮时，前面已经只剩两个数了，这一轮比较完后，下一轮只剩一个数，无需比较

### 2.1 实例

头文件：

```c
#ifndef __BUBBLE_H
#define __BUBBLE_H

//定义函数
void bubble_sort(int array[] ,int length);

#endif

```

源文件：

```c
#include <stdio.h>
#include <stdlib.h>
#include "..\inc\bubble.h"

//参数：数组，数组长度

void bubble_sort(int array[] ,int length)
{
	//先判断传入的数组是否有效
	if(array == NULL || length <= 0)
	{
		printf("数组无效，异常退出\n");
		exit(1);
	}
	//循环length-1轮
	for(size_t i = 0; i < length - 1 ; i ++)
	{
		//每一轮从第一个开始到拍好位置元素之前
		for(size_t j = 0; j < length - 1 - i; j ++)
		{
			if(array[j] > array[j + 1])//如果前面比后面大
			{
				//交换位置
				int temp = array[j];
				array[j] = array[j + 1];
				array[j + 1] = temp;
			}
		}
	}
}
```

主函数文件

```c
#include <stdio.h>
#include "..\inc\bubble.h"

#define SIZE	10
int array[SIZE] = {10 ,15 ,2 ,88 ,16 ,31 ,7 ,4 ,78 ,24};

int main(int argc, const char *argv[])
{
	printf("数组为：10 ,15 ,2 ,88 ,16 ,31 ,7 ,4 ,78 ,24\n");
    printf("冒泡排序后结果为：");//冒泡排序后结果为：2 4 7 10 15 16 24 31 78 88
    bubble_sort(array ,SIZE);
    for (size_t i = 0; i < SIZE; i++)
    {
        printf("%d " ,array[i]);
    }
    printf("\n");

	return 0;
}

```

## 3. 选择排序

以数组为例

假设现在有一个长度为100的int类型数组array\[100]，里面的数大小随机，现在要对其进行从小到大排序（左小右大）

使用选择排序的方法：

*   先假设第一个元素array\[0]为数组中的最小元素 min = array\[0]
*   从第二个元素array\[1]开始到最后一个array\[99]，依次和min比较
*   如果发现有比min更小的 array\[k] ，则更新最小元素 min = array\[k] ,同时记下这个元素的位置 int index = k;
*   以新的最小元素继续后续比较到最后一个array\[99]，这样一轮下来即可找到当前数组中的最小元素索引（每一次有最小的都会记下其索引 index = k）
*   然后把最小元素和最开始选取的元素换位置
*   下一次以下一个元素作为最小元素重复上述操作，一共重复 n - 1 轮即可完成排序	为什么是 n - 1 轮？因为到第99轮时（假设n = 100）就已经只剩下两个元素在比大小了，第100轮则只有一个元素，无需再比较

### 3.1 实例

头文件：

```c
#ifndef __SELECTION_H
#define __SELECTION_H

//定义头文件
void selection_sort(int array[] ,int length);

#endif

```

源文件：

```c
#include <stdio.h>
#include <stdlib.h>
#include "..\inc\selection.h"

//函数参数：数组，数组长度
void selection_sort(int array[] ,int length)
{
	//先判断传入的数组是否有效
	if(array == NULL || length <= 0)
	{
		printf("数组无效，异常退出\n");
		exit(1);
	}
	//从第一个开始循环，循环n - 1轮
	for(size_t i = 0; i < length - 1; i ++)
	{
		int min = array[i];//假设最开始的为最小元素
		//最小值索引假设为当前元素索引
		int index = i;
		//内层循环从第i+1个到最后一个
		for(size_t j = i+1 ;j < length ;j ++)
		{
			//如果后面的有比当前min更小的
			if(array[j] < min)
			{
				//更新最小值
				min = array[j];
				//记下当前最小值的索引
				index = j;
			}
		}
		//内层比较完后得出本轮最小元素的位置，交换其和最开始取出元素的位置，这样本轮最小元素就到数组的左边去了
		int temp = array[i];
		array[i] = array[index];
		array[index] = temp;
	}
	
}
```

主函数文件：

```c
#include <stdio.h>
#include "..\inc\selection.h"

#define SIZE	10
int array[SIZE] = {10 ,15 ,2 ,88 ,16 ,31 ,7 ,4 ,78 ,24};

int main(int argc ,const char *argv[])
{
	printf("数组为：10 ,15 ,2 ,88 ,16 ,31 ,7 ,4 ,78 ,24\n");
    printf("选择排序后结果为：");//选择排序后结果为：2 4 7 10 15 16 24 31 78 88
    selection_sort(array ,SIZE);
    for (size_t i = 0; i < SIZE; i++)
    {
        printf("%d " ,array[i]);
    }
    printf("\n");	

	return 0;
}
```

## 4. 插入排序

以数组为例

假设现在有一个长度为100的int类型数组array\[100]，里面的数大小随机，现在要对其进行从小到大排序（左小右大）

使用插入排序的方法：

*   插入排序将数组分为已排序区间和未排序区间，初始时，已排序区间为空，未排序区间从数组第一个元素array\[0]到最后一个元素 array\[99]
*   取出当前未排序区间第一个数array\[0]，和前面已排序区间中的数进行比较，当前已排序区间为空，则array\[0]直接为已排序区间的第一个数，未排序区间从array\[1]开始
*   取出当前未排序区间第一个数array\[1]，和已排序区间的数进行比较，当前已排序区间只有array\[0]，如果array\[1] < array\[0]则array\[0]后移一位，array\[1]放在其前面，反之直接把array\[1]插入在array\[0]后面 ，此时未排序区间从array\[2]开始
*   取出当前未排序区间第一个数array\[2]，和已排序区间的数进行比较，当前已排序区间有array\[0] (假如上一次交换了位置，则放的是原来array\[1] 的内容)和array\[1]，用array\[2]和array\[1]、array\[0]依次比较
*   假如array\[2] < array\[1]，array\[1]后移一位，array\[2]再和array\[0]比较，如果array\[2] > array\[0] 则array\[2]直接插入在array\[0]后面，在array\[1]前面
*   按照该方式重复，直到排序完成

### 4.1 实例

头文件：

```c
#ifndef __INSERTION_H
#define __INSERTION_H

//函数声明
void insertion_sort(int array[] ,int length);

#endif
```

源文件：

```c
#include <stdio.h>
#include <stdlib.h>
#include "..\inc\insertion.h"

//函数参数：数组，数组长度
void insertion_sort(int array[] ,int length)
{
	//先判断传入的数组是否有效
	if(array == NULL || length <= 0)
	{
		printf("数组无效，异常退出\n");
		exit(1);
	}
	//每一次选取未排序区间的第一个数，刚开始是数组的第一个元素
	int current = 0 ,k = 0;
	for(size_t i = 0; i < length ;i ++)
	{
		current = array[i];
		//已排序区间位置标记
		k = i - 1;//已排序区间最后一个就是未排序区间前一个
		while(current < array[k] && k >= 0)//如果当前未排序数小于已排序数且还有没比较的数
		{
			//当前已排序数后移一位
			array[k + 1] = array[k];
			//再往前比较
			k --;
		}
		//如果遇到大于自己的或者已经比较完成了，则把当前未排序数放到已排序区间的新位置中
		array[k + 1] = current;
	}
}

```

主函数文件：

```c
#include <stdio.h>
#include "..\inc\insertion.h"

#define SIZE	10
int array[SIZE] = {10 ,15 ,2 ,88 ,16 ,31 ,7 ,4 ,78 ,24};

int main(int argc ,const char *argv[])
{
	printf("数组为：10 ,15 ,2 ,88 ,16 ,31 ,7 ,4 ,78 ,24\n");
    printf("插入排序后结果为：");//插入排序后结果为：2 4 7 10 15 16 24 31 78 88
    insertion_sort(array ,SIZE);
    for (size_t i = 0; i < SIZE; i++)
    {
        printf("%d " ,array[i]);
    }
    printf("\n");	

	return 0;
}
```

## 5. 快速排序

以数组为例

假设现在有一个长度为100的int类型数组array\[100]，里面的数大小随机，现在要对其进行从小到大排序（左小右大）

使用快速排序的方法：

*   快速排序使用的是二分的方式，选取一个数作为轴，比这个数小的在左侧，比它大的在右侧，将一个数组经过一次二分后，队剩下的左右侧再次进行二分，循环直到只剩一个数，就完成了排序
*   一般选取给定未排序数组的最右侧元素作为第一次的枢轴，将数组分为左侧区间和右侧区间，初始时左侧区间为空，右侧区间为从数组的第一个元素到枢轴的前一个元素，以array\[100]为例：

    *   枢轴：array\[100]
    *   当前左侧区间最后一个元素的位置：k = -1（空）k是左侧区间位置的标记
    *   当前右侧区间元素范围：array\[0]\~array\[99]
*   从右侧区间第一个元素 array\[0] 开始和枢轴 array\[100] 比较，一直到最后一个 array\[99]
*   如果元素比枢轴大，则左侧区间标记不变，该元素默认就在右侧区间，假设为array\[0]，此时array\[0]就不变，k = -1
*   如果有元素比枢轴小，假设是 array\[1]，则左侧区间位置标记 k+1 留出一个空位，然后把array\[1]的数和array\[k + 1]交换位置，当前情况就是把array\[1]和array\[ -1 + 1 = 0]交换位置，而array\[0]在上一次比较中得出它属于右侧区间，交换后，k = 0，其后一个是右侧区间 ，0+1 = 1 ，因此把array\[0]交换到array\[1]是合理的
*   假设下一个元素 array\[2]还是比枢轴小，同样左侧区间标记 k+1 留出一个空位 ，此时 k = 0+1 = 1，还是将两个元素位置交换 ，array\[2]交换到array\[1]的位置上
*   重复按照上述方式完成左右分区后，假设此时左侧区间标记 k = 15，现在就再把枢轴移动到**左侧区间标记后面的位置**，即交换 array\[100] 和 array\[16]，因为这里k = 15，则其后面一个array\[16]属于右侧区间，因此交换是合理的
*   **最后将枢轴的位置作为返回值返回**

特殊情况：

假设比较的元素从第一个就比枢轴小呢？

*   假设array\[0]比array\[100]小，左侧区间标记 k + 1，交换array\[0]和array\[k + 1]的位置，此时array\[k + 1] = array\[ -1 + 1 = 0] = array\[0]，即array\[0]和array\[0]交换位置，其还是在原位，只是这时候的左侧区间 k = 1，因此交换不一定真的会交换，主要目的是分区

### 5.1 实例

头文件:

```c
#ifndef __QUICK_H
#define __QUICK_H

//函数声明
//交换位置函数
void swap(int *a ,int *b);

//分区函数
int partition(int array[] ,int low ,int high);

//排序函数
void quick_sort(int array[] ,int low ,int high);

#endif
```

源文件：

```c
#include <stdio.h>
#include "..\inc\quick.h"

//交换位置函数
/*
函数参数：要交换的两个数，这里用int *，一方面是因为确定了array[]是int类型的，
二一个方面是因为要想直接交换位置，要么使用返回值去重新赋值
要么就通过传入地址，交换地址里面的内容，
如果是 int a, int b
交换的只是函数形参而非传入的实参
*/
void swap(int *a ,int *b)
{
	int temp = *a;//通过解引用，将a地址中的内容给temp
	*a = *b;//通过解引用，将b地址的内容赋值给a地址中的内容
	*b = temp;//最后再将temp赋值给b地址中的内容完成交换
}

//分区函数
//函数参数：数组，数组的最右侧位置low，数组的最左侧位置high
int partition(int array[] ,int low ,int high)
{
	//选择左侧元素为枢轴
	int pivot = array[high];
	//左侧区间位置标记
	int index = low - 1;//右侧区间前一个
	//从右侧区间第一个数开始比较
	for(size_t i = low ; i < high ; i ++)
	{
		//如果比枢轴小
		if(array[i] < pivot)
		{
			//左侧区间标记+1为新数据流出位置
			index ++;
			//交换位置
			swap(&array[i] ,&array[index]);
		}
		//如果比枢轴大则默认就在右侧区间
	}
	//全部比较完后，将枢轴放在左右区间之间
	swap(&array[high] ,&array[index + 1]);
	//返回枢轴位置
	return (index + 1);
}

//排序函数
void quick_sort(int array[] ,int low ,int high)
{
	//只有当传入的数组存在左右区间时，即有最右侧位置和最左侧位置，且两者不等，才进行分区
	if(low < high)
	{
		//调用分区函数进行初次分区
		//存下分区后的枢轴位置
		int position = partition(array ,low ,high);
		//递归分区左侧区间 low ~ position-1
		quick_sort(array ,low ,(position-1));
		//递归分区右侧区间 position+1 ~ high
		quick_sort(array ,(position+1) ,high);
	}
	
}
```

主函数文件：

```c

#include "..\inc\quick.h"

int main(int argc ,const char *argv[])
{
	printf("数组为：10 ,15 ,2 ,88 ,16 ,31 ,7 ,4 ,78 ,24\n");
    printf("快速排序后结果为：");//快速排序后结果为：2 4 7 10 15 16 24 31 78 88
    quick_sort(array ,0 ,SIZE - 1);
    for (size_t i = 0; i < SIZE; i++)
    {
        printf("%d " ,array[i]);
    }
    printf("\n");

	return 0;
}
```
