# C语法基础

## 1. 位运算复习

### 1.1 按位与 &

有0则为0，全1则为1

```c
unsigned char n1 = 0b11001100;//CC
unsigned char n2 = 0b10111011;//BB

printf("n1 & n2 = %02X (%d) \n" ,n1 & n2 ,n1 & n2);//88 (136)
```

### 1.2 按位或 |

有1为1 ，全0则为0

```c
unsigned char n1 = 0b11001100;//CC
unsigned char n2 = 0b10111011;//BB

printf("n1 | n2 = %02X (%d) \n" ,n1 | n2 ,n1 | n2);//FF (255)
```

### 1.3 按位取反 \~

1变0，0变1

```c
unsigned char n1 = 0b11001100;//CC
unsigned char n2 = 0b10111011;//BB

//按位取反 ~
printf("~n1 = %02X (%d) ~n2 = %02X (%d) \n" ,(char)~n1 ,(char)~n1 ,(char)~n2 ,(char)~n2);//00110011 01000100
```

### 1.4 按位异或 ^

相同为0，不同为1

```c
unsigned char n1 = 0b11001100;//CC
unsigned char n2 = 0b10111011;//BB

//按位异或 ^ 相同为0，不同为1
printf("n1 ^ n2 = %02X (%d) \n" ,n1 ^ n2 ,n1 ^ n2);//01110111
```

### 1.5 左移 <<

不管是否有符号位，低位补0

```c
unsigned char n1 = 0b11001100;//CC
unsigned char n2 = 0b10111011;//BB

//左移 <<
printf("n1 << 1 = %02X (%d) n2 << 1 = %02X (%d) \n" ,(char)(n1 << 1) ,(char)(n1 << 1) ,(char)(n2 << 1) ,(char)(n2 << 1));//10011000 01110110
```

### 1.6 右移 >>

有符号位，如果为负数则高位补1，如果为正数则高位补0，无符号位则高位补0

```c
unsigned char n1 = 0b11001100;//CC
unsigned char n2 = 0b10111011;//BB

//右移 >>
printf("n1 >> 1 = %02X (%d) n2 >> 1 = %02X (%d) \n" ,(char)(n1 >> 1) ,(char)(n1 >> 1) ,(char)(n2 >> 1) ,(char)(n2 >> 1));//01100110 01011101
```

### 1. 7通过位运算实现两个数交换

具体函数：（这里以int类型为例）

```c
void swap(int *a ,int *b)
{
	*a ^= *b;
	*b ^= *a;
	*a ^= *b;
}
```

实例讲解：

```c
char n1 = 0b10101010;   AA
char n2 = 0b11101110;   EE

n1 ^= n2 ---> 01000100  44  0xEE - 0xAA = 0x44 其实这一步就是在求n1和n2的差值
n2 ^= n1 ---> 10101010  AA  这一步实际上就是0xEE - 0x44
n1 ^= n2 ---> 11101110  EE  这一步实际上就是0x44 + 0xAA
```

### 1.8 通过位运算实现求绝对值

首先要知道的是**负数求绝对值是 将原码按位取反然后再+1 ，也等同于先将原码-1然后再整体按位取反** , 而正数的绝对值就是其本身

具体函数：(以int类型为例)

```c
void abs_data(int *data_ad)
{
	//取出符号位
	int mask = *data_ad >> 31;
	//求出绝对值
	*data_ad = (*data_ad + mask) ^ mask;
}
```

实例讲解：

```c
//求绝对值
void abs_data(int *data_ad)
{
    //负数的绝对值为 原码 取反 + 1 也等同于 原码 - 1 然后整体取反，正数绝对值是本身
    //先获得符号，根据有符号数中 负数首位为 1 ，正数首位为 0
    int mask = (*data_ad) >> 31;
    /*
    为什么是右移31位？
    因为int类型刚好是32位，而首位是符号位，右移31位后只剩下符号位
    */
    //绝对值计算
    *data_ad = (*data_ad + mask)^mask;
    /*
    为什么是对mask按位异或而不是直接 ~(*data_ad + mask)
    如果直接整体取反，假设data_ad是一个正数，直接取反会变成负数
    正数时，mask = 0，假设data_ad = 0111 0111 
		(*data_ad + mask)^mask = (0111 0111 + 0) ^ 0000 0000 
                                = 0111 0111还是不变
    负数时，mask = 1，假设data_ad = 1011 0101(-75) 
		(*data_ad + mask)^mask = (1011 0101 + 1111 1111) ^ 1111 1111 
                                = 1011 0100 ^ 1111 1111 
                                = 0100 1011(75)已转为绝对值
    */                                                              
}

```

## 2. 查表法

先定义好要查询的表，一般都是数组形式，然后实际使用的时候直接通过对应的索引来获取

**表中的内容一般都是一个项目中的统一信息提示，因此最好添加 const 修饰，一方面是可以防止表在其他文件被修改，另一方面是编译器会对const修饰的变量做一些优化**

实例：

```c
#include <stdio.h>

//查表法
/*
先定义好要查询的表
要查的表一般定义为 const 类型的，一方面是为了防止表被意外修改，另外一方面是因为编译器可以优化
*/

const char *days[] = {"星期一" ,"星期二" ,"星期三" ,"星期四" ,"星期五" ,"星期六" ,"星期天"};

int main(int argc, char const *argv[])
{
    int day;
    printf("请输入天数：");
    scanf("%d" ,&day);

    if (day % 7 == 0)
    {
        printf("第 %d 天是 %s \n" ,day ,days[6]);
    }
    else
    {
        int temp = day % 7;

        printf("第 %d 天是 %s \n" ,day ,days[temp - 1]);
    }
    
    return 0;
}
```

## 3. 跑马灯（位运算）

这里以八位跑马灯为例，每一位对应一个灯，1代表亮起，0代表熄灭

每次灯组显示之间有小的延时，可通过添加对应头文件实现

```c
#include <unistd.h>

//具体函数
sleep();//参数是秒数

usleep();//参数是微秒数  1s = 1000ms = 1000000ms 1ms= 1000us
```

实例：

```c
#include <stdio.h>
#include <unistd.h>

//灯组长度
#define LIGHT_SIZE	8

int main(int argc ,const *char argv[])
{
	//定义灯组
	int light = 0x01 ;//这里用两位16进制表示八位二进制就是 0000 0001
	//定义灯移动的方向
	int direction = 0;//0从右向左 ，1从左向右

	while(1)
	{
		//使用for循环打印出灯组中每一个灯的亮灭状态
		for(int i = LIGHT_SIZE - 1; i >= 0 ;i --)
		{
			if(light & (1 << i))
			{
				printf("● ");//亮起
			}
			else
			{
				printf("○ ");//熄灭
			}
		}
		printf("\n");
		//延时
		usleep(300000);
		//改变灯组中灯的亮灭状态
		//先判断灯组移动方向
		if(direction == 0)//向左移动
		{
			light <<= 1;//左移一位
			//移动上限
			if(light == 0x80) // 1000 0000 最左边的灯亮起
			{
				//改变灯组移动方向为向右
				direction = 1;
			}
		}
		else//如果是向右移动
		{
			light >>= 1;//右移一位
			//移动上限
			if(light == 0x01) // 0000 00001 最右边的灯亮起
			{
				//改变灯组移动方向为向左
				direction = 0;
			}
		}
	}

	return 0;
}
```

其中 1 << i 是通过将1左移，然后再和light进行相与，去判断对应的位是否为1 ，如果是，则该位亮起，反之则不亮

例如

```c
//开始时

(1<<i) = 1000 0000;//左移七位
light & (1 << i) = 0000 0000;//说明最左侧的灯不亮
```

多个数码管显示

```c
void multi_seg(int input_num)
{
	//要把一串数字一个一个拆分，因此需要一个数组来存每一个数字
	int nums[10];
	/*
	为什么是10位？
	int类型最大就是10位
	*/
	//用一个计数器来计数实际有多少位，同时也是数组中的索引
	int counter = 0;
	//先判断传入的是否为0
	if(input_num == 0)
	{
		nums[counter ++] = 0;
	}
	else
	{
		//连续对10取余和整除10从个位取出每一位
		while(input_num != 0)
		{
			nums[counter ++] = input_num % 10;
			input_num /= 10;
		}
	}
	//倒序在数码管中显示
	for(int i = counter - 1 ; i >= 0; i--)
	{
		signal_seg(nums[i] ,counter - 1 - i);
	}
}
```

	{
		printf("不能为负数\n");
	}

	multi_seg(input);

	return 0;
}
