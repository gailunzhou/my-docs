# C语法基础

## 1. if 选择结构

### 1.1 基本if选择结构

语法:

```c
#include <stdio.h>

int main()
{
	if(条件)
	{
		执行语句;
	}
	return 0;
}
```

**说明:**

*   条件只能是bool类型的值或者表达式

执行的语句可以是一组语句：

```c
#include <stdio.h>

int main(int argc, char const *argv[])
{
    int score = 0;
    printf("请输入成绩 \n");
    scanf("%d" ,&score);

    if(score < 0 || score >100)
    {
        printf("成绩错误，请再次输入\n");
		scanf("%d" ,&score);
    }

	return 0;
}
```

## 1.2 if...else...结构

语法：

```c
#include <stdio.h>

int main()
{
	if(条件)
	{
		执行语句1;
	}
	else
	{
		执行语句2;
	}

	return 0;
}
```

**说明：**

*   和三目运算本质相同

举例：

```c
#include <stdio.h>

int main(int argc, char const *argv[])
{
    int score = 0;
    printf("请输入成绩 \n");
    scanf("%d" ,&score);

    if (score > 60)
    {
        printf("及格\n");
    }
    else
    {
        printf("不及格\n");
    }

	return 0;
}
```

## 1.3 if...else if...else... 结构

语法：

```c
#include <stdio.h>

int main()
{
	if(条件)
	{
		执行语句1;
	}
	else if(条件)
	{
		执行语句2;
	}	
	else
	{
		执行语句3;
	}

	return 0;
}
```

**说明:**

*   所有的分支只能满足一个，条件互斥，因此能执行的语句只有一个

例如：

```c
#include <stdio.h>

int main(int argc, char const *argv[])
{
    int score = 0;
    printf("请输入成绩 \n");
    scanf("%d" ,&score);

    // if...else if...else 只能按照程序所写的顺序去判断
    if(score > 90)
    {
        printf("优\n");
    }
    else if(score > 80 && score < 90)
    {
        printf("良\n");
    }
    else if(score > 70 && score < 80)
    {
        printf("中\n");
    }
    else if(score > 60 && score < 70)
    {
        printf("及格\n");
    }
    else
    {
        printf("不及格！\n");
    }

    return 0;
}

```

## 2.switch选择结构

语法:

```c
#include <stdio.h>

int main()
{
	switch(key)
	{
		case:
			执行语句1;
	}
	
	return 0;
}
```

**说明：**

*   每个case之间顺序可以交换 不影响代码运行
*   default不一定放在末尾 在所有case失效的时候就会被执行到
*   case中值与 key类型必须一致
*   匹配到条件，执行case遇到break退出
*   如果没有break，从上到下执行case，直到遇到break&#x20;
*   没有匹配到条件执行default遇到break退出，一个都没有匹配上执行到最后结束。
*   如果没有break，从上到下执行case，直到遇到break，一个都没有匹配上执行到最后结束

## 2. 循环

### 2.1 while

语法:

```c
#include <stdio.h>

int main
{
	while(条件)
	{
		执行语句;
	}
	return 0;
}
```

举例:

```c
#include <stdio.h>

//while

int main(int argc, char const *argv[])
{
    int i = 0;
    //简单使用
    while (i < 10)
    {
        printf("%d ", i);
        i++;
    }
    printf("\n");

    //打印偶数
    i = 0;
    while (i < 10)
    {
        if(i % 2 == 0)
        {
            printf("%d ", i);
        }
        i++;
    }
    printf("\n");

    return 0;
}

```

### 2.2 do-while

语法：

```c
#include <stdio.h>

int main
{
	do
	{
		执行语句;
	}
	while(条件)

	return 0;
}
```

**说明：**

*   首先执行do后面结构体中的执行语句
*   判断循环条件是否为true，如果true，执行do后执行语句，如果为false终止循环

举例：

```c
#include <stdio.h>

int main(int argc, char const *argv[])
{
    /* code */
    //do{}while();和while(){}比会至少执行一次，而while存在一次都不执行的情况
    int i = 0;
    
    do
    {
        i++;
        printf("i = %d \n" ,i);
        printf("执行了\n");
    } while (i < 0);
    
    i = 0;
    while (i < 0)
    {
        i++;
        printf("i = %d \n" ,i);
        printf("执行了\n");
    }
    

    return 0;
}
```

### 2.3 for循环

语法：

```c
#include <stdio.h>

int main
{
	for(初始化条件A ;判断条件B ;循环后语句C )
	{
		执行语句;
	}

	return 0;
}
```

使用举例：

```c
#include <stdio.h>

int main(int argc, char const *argv[])
{
    /* code */
    //for( ; ; ){}
    for (size_t i = 0; i < 10; i++)
    {
        /* code */
        if(i % 2 == 0)
        {
            printf(" i = %d " ,i);
        }
    }
    printf("\n");

    //用for循环计算1-10的 积(即10的阶乘)1\*2\*3\*。。。10
    int j = 1;
    for (size_t i = 1; i <= 10; i++)
    {
        j *= i;
        printf("第 %d 次乘积 结果为 %d \n" ,i ,j);
    }

    return 0;
}

```

### 2.4 goto

语法：

```c
#include <stdio.h>

int main
{
	lableA:
		...
	goto lableA;
		...

	return 0;
}
```

**说明：**

*   goto无条件跳转
*   向后的goto可以跳出循环
*   向前的goto可以进入循环
*   **慎用！容易进入死循环且降低代码可读性！**

举例:

```c
#include <stdio.h>

int main(int argc, char const *argv[])
{
    /* code */
    //goto 无条件跳转

    printf("step0 \n");

    goto A;

    B:

    printf("step1 \n");

    A:
    
    printf("step2 \n");
    printf("step3 \n");

    //通过往后goto可进入循环
    goto B;

    printf("step4 \n");

    //通过往前goto可跳出循环
    while (1)
    {
        printf("这是while死循环 \n");
        goto C;
    }
    
    C:

    printf("这是while死循环外面！ \n");

    return 0;
}

```

### 2.5 死循环

三个循环的死循环:

```c
#include <stdio.h>

int main(int argc, char const *argv[])
{
    /* code */
    //死循环
    //1.while
    while (1)
    {
        printf("这是while死循环 \n");
    }

    //2.do while
    do
    {
        printf("这是do-while死循环 \n");
    } while (1);
    
    //3.for
    for (; ;)
    {
        printf("这是for死循环 \n");
    }

    return 0;
}

```

### 2.6 循环控制关键字

分为：

*   continue
*   break
*   return

使用举例：

```c
#include <stdio.h>

int main(int argc, char const *argv[])
{
    /* 
    循环控制语句
    */
    //1.break   该关键字触发后，其后代码不执行，直接跳出循环
    printf("========break跳出循环========\n");
    int i = 0;
    while (1)
    {
        i ++;
        if(i == 10)
        {
            break;
        }
        printf("循环了%d次 \n" ,i);
    }
    printf("跳出循环了！\n");

    //2.continue  该关键字触发后，其后代码不执行，直接执行下一次循环
    printf("========continue跳过循环========\n");
    int j = 0;
    while (1)
    {
        j ++;
        if(j == 5)
        {
            continue;
        }
        if(j == 10)
        {
            break;
        }
        printf("循环了%d次 \n" ,j);
    }
    printf("跳出循环了！\n");

    //3.return  该关键字触发后，其后代码不执行，直接退出函数
    printf("========return跳出循环========\n");
    int k = 0;
    while (1)
    {
        k ++;
        if (k == 10)
        {
            return 10;
        }
        printf("循环了%d次 \n" ,k);
    }
    printf("跳出循环了！\n");

    return 0;
}

```

### 2.7 嵌套循环

主要是for循环的应用:

```c
#include <stdio.h>

int main(int argc, char const *argv[])
{
    /* 
    循环里面嵌套循环 
    */

    /*
    嵌套for循环：
    外层循环用于确定层数
    内层循环用于确定每一层的内容
    */

    /*

    用for循环打印出一个
    *
    * *
    * * *
    * * * *
    * * * * *

    */
   for (size_t i = 1; i <= 5; i++)
   {
        for (size_t j = 1; j <= i; j++)
        {
            printf("* ");
        }
        
    printf("\n");
   }

   /*

    用for循环打印出一个
        *
       * *
      * * *
     * * * *
    * * * * *

    */
   for (size_t i = 5; i >= 1; i--)
   {
    for (size_t j = i-1; j > 0; j--)
    {
        printf(" ");
    }
    for (size_t k = 1; k <= 6-i; k++)
    {
        printf("* ");
    } 
    printf("\n");
   }

   //解法2
   for (size_t i = 1; i < 6; i++)//确定5层
   {
    for (size_t j = 5; j > i; j--)//打印每一层前面的空格
    {
        printf(" ");
    }
    for (size_t k = 1; k <= i; k++)//打印*
    {
        printf("* ");
    }
    printf("\n");
   }
   
   /*
    用for循环打印出99乘法表
    */
   for (int i = 1; i <= 9; i++)
   {
    for (int j = 1; j <= i; j++)
    {
        printf("%d x %d = %d\t" ,j ,i ,j*i);
    }
    printf("\n");
   }

    return 0;
}

```
