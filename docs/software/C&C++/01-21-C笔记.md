# C数据结构

## 1. 线性数据结构

### 1.1 队列

*   队列Queue是一种特殊的线性表，**队列只允许在表的一端进行插入操作，在表的另一端进行删除操作；**
*   **进行插入操作的一端为队尾，进行删除操作的一端为队首**，队列中没有数据元素时，称之为空队列。
*   **队列操作的规则：先进先出FIFO**
*   **通常情况下，队列的数据在入队(enqueue)和出队(dequeue)操作时不允许修改**。队列的目的是按照先进先出(FIFO)的顺序管理数据，如果允许修改队列中的数据，可能会导致队列的顺序性被破坏。

可用数组或者链表实现，下面以链表实现为例

#### 1.1.1 队列创建

队列节点结构体

```c
//定义队列节点结构体
```

队列结构体


#### 1.1.2 入队

```c
//入队---添加到队尾
void enqueue(queue *queue_ad ,void *data_ad)
{
    if (queue_isok(queue_ad))
    {
        //创建一个新队列节点
        queue_node *new_node = create_node(data_ad);
        //如果当前队列为空
        if (queue_ad->rear == NULL)
        {
            //则队首和队尾都是新节点
            queue_ad->front = new_node;
            queue_ad->rear = new_node;
        }
        else
        {
            //如果有其他节点，队尾的next = 新节点
            queue_ad->rear->next = new_node;
            //更新队尾
            queue_ad->rear = new_node;
        }
        //队列数量+1
        queue_ad->size ++;

        printf("已入队\n");
        return ;
    }
    printf("队列无效\n");
}
```

#### 1.1.3 出队

```c
//出队---从队首出
void *dequeue(queue *queue_ad)
{
    if (queue_isok(queue_ad))
    {
        if (queue_ad->rear == NULL)
        {
            printf("队列为空\n");
            return NULL;
        }
        //获取队首节点
        queue_node *temp = queue_ad->front;
        //获取数据
        void *data = temp->data;
        //判断是否只剩一个数据
        if (queue_ad->size == 1)
        {
            queue_ad->front = NULL;
            queue_ad->rear = NULL;
        }
        else
        {
            //更新队首
            queue_ad->front = queue_ad->front->next;
        }

        //释放出队节点
        free(temp);
        temp = NULL;

        //队列数量-1
        queue_ad->size --;

        //返回数据
        return data;

    }
    printf("队列无效\n");
    return NULL;
}
```

#### 1.1.5 释放队列

```c
//释放队列
void free_queue(queue *queue_ad)
{
    printf("开始释放\n");
    if (queue_isok(queue_ad))
    {
        //从队首开始
        queue_node *current = queue_ad->front;
        while (current != NULL)
        {
            queue_node *temp = current;
            current = current->next;
            free(temp);
        }
        //释放队列
        queue_ad->front = NULL;
        queue_ad->rear = NULL;
        queue_ad->size = 0;

        free(queue_ad);

        printf("队列已释放\n");
        return ;
    }
    printf("队列无效\n");
}
```

### 1.2 哈希表

数组 + 链表

每一个数组存储着一个键key所对应的链表的首节点地址

为什么有链表？

因为假设第一次存入的key为1，第二次存入的key为11，经过hash算法得出的位置会出现冲突（在同一个位置），因此需要链表来存储后续冲突的节点

这里简单示例：

#### 1.2.1 哈希表的创建

链表节点结构体

```c
#define HASH_TABLE_SIZE     10
//hash表节点结构体
typedef struct hash_node
{
    //键域
    int key;
    //值域
    int value;
    //指针域
    struct hash_node *next;
}hash_node;
```

哈希表初始化

```c
#include <stdlib.h>

//hash表
hash_node *hash_table[HASH_TABLE_SIZE];

//初始化hash表
void hash_table_init(void)
{
    for (size_t i = 0; i < HASH_TABLE_SIZE; i++)
    {
        hash_table[i] = NULL;
    }
    
}
```

哈希函数

哈希函数就是用来计算存储到数组的哪个索引的一种算法，有很多，这里以保留除数法为例

又称质数除余法，设Hash表空间长度为m，选取一个不大于m的最大质数p，  H(key)=key%p

```c
int hash_function(int key)
{
    return key % HASH_TABLE_SIZE;
}
```

#### 1.2.2 插入键值对到哈希表中

```c
//插入键值对到hash表中
void insert_to_hash(int key ,int value)
{
    if (key < 0 || value < 0)
    {
        printf("键或值不能为负\n");
        return ;
    }
    //新建节点
    hash_node *new = (hash_node *)calloc(1 ,sizeof(hash_node));
    if (new == NULL)
    {
        printf("节点创建失败\n");
        return ;
    }
    new->key = key;
    new->value = value;
    new->next = NULL;

    int index = hash_function(key);
    if (Table[index] == NULL)
    {
        Table[index] = new;
        printf("插入键值对成功\n");
    }
    else
    {
        hash_node *current = Table[index];
        while (current->next != NULL)
		/*
			为什么是current->next != NULL不是current != NULL？
			因为前者是找到最后一个节点，后者是找到最后一个节点后的空位，
			此处如果选择后者，current = NULL，然后后面的current -> next是没有任何地址的，就无法实现赋值
		*/
        {
            current = current->next;
        }//找到哈希表中链表末尾的位置
        current->next = new;
        printf("插入键值对成功\n");
    }

}
```

#### 1.2.3 根据键获取对应的值

```c
//根据key查找对应的值
int search_data(int key)
{
    if (key < 0)
    {
        printf("键不能为负\n");
        return -1;
    }
    int index = hash_function(key);
    hash_node *current = Table[index];

    while (current != NULL)
	/*
	这里为什么是 current != NULL 而不是 current -> next != NULL?
	因为如果选择后者，当查找的是最后一个的时候，最后一个节点 current -> next = NULL，
	不会进入内部的 return current->value 而是直接退出到 printf("表中没有目标键\n");
	*/
    {
        if (current->key == key)
        {
            return current->value;
        }
        current = current->next;
    }
    printf("表中没有目标键\n");
}

```

## 2. 非线性结构

### 2.1 树（红黑树）

是一种特化的AVL树（**平衡二叉树：每个结点的左子树和右子树的高度差至多等于1**），都是在进行插入和删除操作时通过特定操作保持二叉查找树的平衡，从而获得较高的查找性能。

**​维护树平衡手段 1. 左旋/右旋    2.重新着色**

红黑树是一种自平衡的二叉查找树，具有以下五个关键特性：

​	**1.节点颜色**：每个节点都是红色或黑色。

​	**2.根节点**：根节点必须是黑色。

​	**3.叶子节点**：所有的叶子节点（即空节点）都是黑色。

​	**4.红色节点的约束**：红色节点的子节点必须是黑色（即红色节点不能有红色子节点）。

​	**5.黑色节点的平衡**：从任一节点到其每个叶子节点的所有路径都包含相同数目的黑色节点。

​	这些特性确保了红黑树的平衡性，使得其在最坏情况下的时间复杂度仍然保持在 O(logN)。具体来说，红黑树通过以下方式实现自平衡：

​		**旋转操作**：包括左旋和右旋，用于调整节点的位置以维持树的平衡。

​		**变色操作**：用于调整节点的颜色以满足红黑树的特性。

#### 2.1.2 修复操作

左旋操作

```c
//左旋操作
/*
    \ /            \ /  (意为 \ 或者 / )
     x              y
    / \            / \
   a   y   --->   x   c
      / \        / \   
     b   c      a   b
*/
void left_rotate(tree *tree_ad ,tree_node *target_node)
{
    //设置y为x的右子节点
    tree_node *y = target_node->rchild;
    //设置x的右子节点为y的左子节点
    target_node->rchild = y->lchild;
    //判断y的左子节点是否为哨兵节点
    if (y->lchild != tree_ad->nil)
    {
        //如果不是，则设置y的左子节点的父节点为x
        y->lchild->parent = target_node;
    }
    /*
    为什么要先将x的右子节点设置为y的左子节点，再来确认是否为哨兵？
    因为哨兵节点不用赋值父级节点，而非哨兵节点是需要确定父级节点的，
    但是非哨兵节点的子节点是哨兵节点不需要哨兵节点进行同意
    */
    //将y的父级节点更改为x的父级节点
    y->parent = target_node->parent;
    //确认x是否为根节点
    if (target_node->parent == tree_ad->nil)//如果是根节点
    {
        //更新树的根节点
        tree_ad->root = y;
    }
    //再确认x是其父级节点的左子节点还是右子节点
    else if (target_node->parent->lchild == target_node)//如果是其父级节点的左子节点
    {
        target_node->parent->lchild = y;//则更新为y
    }
    else//如果是右子节点
    {
        target_node->parent->rchild = y;
    }
    //将y的左子节点换为x
    y->lchild = target_node;
    //再设置x的父级节点为y
    target_node->parent = y;
}
```

右旋

```c
//右旋操作
/*
     \ /          \ / (意为 \ 或者 / )
      y            x
     / \          / \
    x   c  --->  a   y
   / \              / \
  a   b            b   c
*/
void right_rotate(tree *tree_ad ,tree_node *target_node)
{
    //设置x为y的左子节点
    tree_node *x = target_node->lchild;
    //先把y的左子节点赋值为x的右子节点
    target_node->lchild = x->rchild;
    //判断x的右子节点是否为哨兵节点
    if (x->rchild != tree_ad->nil)
    {
        //如果不是则设置其父级节点为y
        x->rchild->parent = target_node;
    }
    //先把x的父级节点改为y的父级节点
    x->parent = target_node->parent;
    //判断y是否为根节点
    if (target_node->parent == tree_ad->nil)
    {
        //更新根节点为x
        tree_ad->root = x;
    }
    else if (target_node->parent->lchild == target_node)//如果y是其父级节点的左子节点
    {
        target_node->parent->lchild = x;
    }
    else
    {
        target_node->parent->rchild = x;
    }
    //设置x的右子节点为y
    x->rchild = target_node;
    //设置x为y的父级节点
    target_node->parent = x;
}
```

完整修复操作

```c
//修复操作
void fixup(tree *tree_ad ,tree_node *target)
{
    //新节点自己是红色，如果父级节点也是红色则需要修复
    //因为情况1是需要继续向上判断，所以此处是while
    while (target->parent->color == RED)
    {
        printf("需要修复，开始修复\n");
        //先判断父级节点是祖父节点的左子节点还是右子节点，然后再判断叔叔节点是不是红色，最后再判断自身是父级节点的左子节点还是右子节点
        /*
        为什么先判断这条，然后再去判断叔叔节点的颜色，再去判断自身的位置？
        因为先判断这条，可以区分出是三种情况的基础情况还是镜像情况
        然后判断叔叔节点的颜色是为了知道是情况1还是情况2
        最后才去判断自身是父级节点的左子节点还是右子节点可以区分出情况2还是情况3
        */
        if (target->parent->parent->lchild == target->parent)//如果祖父节点的左子节点是父级节点
        {
            //判断叔叔节点的颜色
            tree_node *y = target->parent->parent->rchild;
            //叔叔节点为祖父节点右子节点
            if (y->color == RED)//红色
            {
				/*
				情况1：祖父节点为黑色，父级节点和叔叔节点为红色，新节点为红色，两红相遇
				*/
                printf("修复：情况1\n");
                //将父级节点和叔叔节点都改为黑色
                target->parent->color = BLACK;
                y->color = BLACK;
                //将祖父节点改为红色
                target->parent->parent->color = RED;
                //将祖父节点作为新的子节点往上判断
                target = target->parent->parent;
            }
            else//如果叔叔节点为黑色
            {
                //判断自身节点的位置确定是情况2还是情况3
                if (target->parent->rchild == target)//如果是右节点
                {
					/*
					情况2：祖父节点为黑色，父级节点为红色，叔叔节点为黑色，新节点为红色，新节点和父级节点的关系和父级节点和祖父节点的关系不同，两红相遇
					*/
                    printf("修复：情况2\n");
                    //将子节点更改为父级节点
                    target = target->parent;
                    //左旋父级节点
                    left_rotate(tree_ad ,target);
                    /*
                    为什么是先改节点，再左旋？
                    假设先左旋，令target的父级节点叫father
                    左旋father后，target会变成father的父级节点，
                    而father会变成target的左子节点
                    而此时再按照情况3去右旋祖父节点，是有问题的，
					因为情况3说的祖父节点是此时father的祖父节点，
                    同时也是左旋后target的父级节点，因为情况3是按照插入时为
                                    grandpa(黑色)
                                     /
                                  father（红色）
                                   /
                                target（红色）
                    的情况来修复，而先左旋之后的实际情况是
                                    grandpa(左旋后target的father)
                                     /
                                  target
                                   /
                                father
                    而解决情况2的方法是转换为情况3，因此要先修改target = target->parent 再左旋
                    */
                }
                //如果是左子节点
				/*
				情况3：祖父节点为黑色，父级节点为红色，叔叔节点为黑色，新节点为红色，新节点和父级节点的关系和父级节点和祖父节点的关系相同 ，两红相遇
				*/
                printf("修复：情况3\n");
                //修改祖父节点颜色为红色
                target->parent->parent->color = RED;
                //修改父级节点颜色为黑色
                target->parent->color = BLACK;
                //右旋祖父节点
                right_rotate(tree_ad ,target->parent->parent);
                /*
                为什么要先改颜色再左/右旋？
                因为左/右旋后父子节点关系会交换，此时再改颜色改的就不是正确目标的颜色
                */
            }
            
        }
        else//如果祖父节点的右子节点是父级节点
        {
            //判断叔叔节点的颜色区分情况1还是情况2
            tree_node *y = target->parent->parent->lchild;
            //叔叔节点为祖父节点左子节点
            if (y->color == RED)
            {
                printf("情况1：镜像\n");
                //将父级节点和叔叔节点颜色改为黑色
                target->parent->color = BLACK;
                y->color = BLACK;
                //将祖父节点颜色改为红色
                target->parent->parent->color = RED;
                //将祖父节点改为判断节点往上进行判断
                target = target->parent->parent;
            }
            else//如果叔叔节点为黑色
            {
                //判断自身位置区分情况2和情况3
                if (target->parent->lchild == target)//如果自身是左子节点
                {
                    printf("情况2：镜像\n");
                    //先修改判断节点为父级节点，给情况3做准备
                    target = target->parent;
                    //再右旋父级节点
                    right_rotate(tree_ad ,target);
                }
                //如果插入时就为右子节点
                printf("情况3：镜像\n");
                //祖父节点改为红色
                target->parent->parent->color = RED;
                //父级节点改为黑色
                target->parent->color = BLACK;
                //左旋祖父节点
                left_rotate(tree_ad ,target->parent->parent);
            }
            
        }
        printf("修复完成\n");
    }
    //根节点一定为黑色
    tree_ad->root->color = BLACK;
}
```

插入函数

```c
//插入操作
void insert_node(tree *tree_ad ,void *data_ad ,int (*func)(void *A ,void *B))
{
    //新建节点
    tree_node *new_node = create_treenode(tree_ad ,data_ad);
    //先声明父级节点为哨兵节点
    tree_node *father = tree_ad->nil;
    /*
    为什么要单独用一个变量来存放父级节点，经过下面的遍历不是能找到该插入在哪个节点下面吗？
    因为下面的逻辑是判断current != tree_ad->nil 不为哨兵节点才结束遍历，也即是说遍历结束时，current = tree_ad->nil，而不是找的正确节点.
    */
    //从根节点开始比较
    tree_node *current = tree_ad->root;
    while (current != tree_ad->nil)//一直比较到哨兵节点
    {
        //先把比较的节点赋值给父级节点
        father = current;
        if (func(new_node->data ,current->data) < 0)//小于则放左侧
        {
            current = current->lchild;
        }
        else//大于则放右侧
        {
            current = current->rchild;
        }
        
    }
    //确定新节点的父级节点
    new_node->parent = father;
    //再判断新节点的位置
    if (tree_ad->root == tree_ad->nil)//如果为空树
    {
        tree_ad->root = new_node;//新节点为根节点
    }
    else if (func(new_node->data ,father->data) < 0)//小于父级节点则在为左子节点
    {
        father->lchild = new_node;
    }
    else//大于则为右子节点
    {
        father->rchild = new_node;
    }
    //插入后新节点默认为红色，可能父级节点也为红色，需要修复
    printf("插入成功，开始确认是否需要修复\n");
    fixup(tree_ad ,new_node);
}
```

```c
//先序遍历
void preorder_traversal(tree *tree_ad ,tree_node *target ,void (*func)(tree_node *node))
{
    if(target != tree_ad->nil)//如果不为哨兵节点
    {
        //中
        func(target);
        //左 递归
        preorder_traversal(tree_ad ,target->lchild ,func);
        //右 递归
        preorder_traversal(tree_ad ,target->rchild ,func);
```

```c
        if (node_ad != tree_ad->nil)

    }
    
}
```
