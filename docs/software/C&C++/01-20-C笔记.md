# C数据结构

## 1. 链表

**顺序存储由于存储单元在物理上必须都是相邻的**，主要有以下两大缺点：

*   **插入和删除操作需要移动大量元素**
*   **采用顺序存储必须先分配内存单元，提前定义线性表的长度，但是实际分配的大小可能并不合适**

因此使用**存储单元在物理上可以并不相邻的链表**进行解决

同样，**顺序存储在数据的检索上相较于链表更快，数据越多，链表检索数据越慢**

### 1.1 链表结构

**链表中每个元素由两部组成：**

*   **数据域**
*   **指针域（指向上一个或者下一个元素）**

因此有单向链表和双向链表和环形链表等：（prev表示上一个元素的地址，next表示下一个元素的地址）

*   单向链表：每一个元素只能单独指向上一个元素或者单独指向下一个元素，链表的尾部元素指向NULL
*   双向链表：每一个元素都可以同时指向上一个元素和下一个元素，链表的头部元素的prev为NULL，尾部元素的next为NULL
*   环形链表：链表的头部元素的prev为尾部元素，尾部元素的next为头部元素

下面以双向链表为例

定义链表中节点的结构体

```c
typedef struct list_node
{
	//数据域
	void *data;
	//指针域
	struct list_node *prev;//前驱指针
	struct list_node *next;//后继指针
}list_node;
```

定义链表结构体

```c
typedef struct list
{
	//指针域
	list_node *head;//头部指针
	list_node *tail;//尾部指针
	//链表元素计数
	int size;
}list;
```

创建一个节点

```c
#include <stdlib.h>

list_node *create_node(void *data_ad)

	{
		printf("节点创建失败，内存分配失败\n");
		return NULL;
	}
	new_node->data = data_ad;
	new_node->prev = NULL;
	new_node->next = NULL;

	return new_node;
```

释放链表

```c
void free_list(list *list_ad)
{
    if (list_ad == NULL)
    {
        printf("目标链表为空链表\n");
        return ;
    }
    //先释放节点
    list_node *current = list_ad->head;//从表头开始
    while (current != NULL)
    {
        list_node *temp = current;
        current = current->next;//下一个节点

        free(temp);//释放当前节点
    }
    //释放链表
    list_ad->head = NULL;
    list_ad->tail = NULL;
    list_ad->size = 0;

    free(list_ad);
    list_ad = NULL;
    
    printf("已释放\n");
}
```

#### 1.1.1 链表的增

在头部插入新节点

```c
void prev_headnode(list *list_ad ,void *data)
{
    //先判断链表是否有效
    if (list_isok(list_ad))
    {
        //创建一个新节点
        list_node *new = create_node(data);
        //判断链表是否为空
        if (list_ad->head == NULL)//头节点为空则说明整个链表都为空
        {
            list_ad->head = new;//头节点为新节点
            list_ad->tail = new;//尾节点也为新节点
        }
        else
        {
            //新节点next = 当前头节点
            new->next = list_ad->head;
            //当前头节点prev = 新节点
            list_ad->head->prev = new;
        }
        //更新当前头节点
        list_ad->head = new;
        //链表数量+1
        list_ad->size ++;

        printf("头部插入节点成功\n");
        return;
    }
    printf("链表无效\n");
}
```

在某项节点后插入新节点

```c
void insert_node(list *list_ad ,list_node *target_node ,void *data)
{
    if (list_isok(list_ad))
    {
        if (target_node == NULL)
        {
            printf("插入节点失败，前驱节点不能为空\n");
            return ;
        }
        else
        {
            //创建一个新节点
            list_node *new = create_node(data);
            //新节点next = 前驱节点next
            new->next = target_node->next;
            //新节点prev = 前驱节点
            new->prev = target_node;

            //是否有后继节点
            if (target_node->next == NULL)//无
            {
                //前驱节点next = 新节点
                target_node->next = new;
                //更新尾节点
                list_ad->tail = new;
            }
            else//有后继节点
            {
                //前驱节点next = 新节点
                target_node->next = new;
                //后继节点prev = 新节点
                target_node->next->prev = new;
            }
            //节点数+1
            list_ad->size ++;
            printf("节点插入成功\n");
            return ;
        }
        
    }
    printf("链表无效\n");
}

```

#### 1.1.3 链表的删

```c
//删除链表中某个节点
void delete_node(list *list_ad ,list_node *target_node)
{
    if (list_isok(list_ad))
    {
        if (list_isempty(list_ad))
        {
            printf("目标链表为空\n");
            return ;
        }
        if (list_ad->size != 1)//不为一个
        {
            //如果删除的是头节点
            if (list_ad->head == target_node)
            {
                //target_node的后继节点的前驱节点为NULL
                target_node->next->prev = NULL;
                //更新链表头节点
                list_ad->head = target_node->next;
            }
            else if (list_ad->tail == target_node)//如果删除的是尾节点
            {
                //target_node的前驱节点的后继节点为NULL
                target_node->prev->next = NULL;
                //更新尾节点
                list_ad->tail = target_node->prev;
            }
            else//如果是其他节点
            {
                //target_node的前驱节点的后继节点 = target_node的后继节点
                target_node->prev->next = target_node->next;
                //target_node的后继节点的前驱节点为 = target_node的前驱节点
                target_node->next->prev = target_node->prev;
            }
        }
        else
        {
            list_ad->head = NULL;
            list_ad->tail = NULL;
        }
        
        //释放目标节点
        target_node->data = NULL;
        target_node->next = NULL;
        target_node->prev = NULL;

        free(target_node);
        //链表节点数--
        list_ad->size --;
        printf("已删除目标节点\n");
        return ;
    }
    printf("链表无效\n");
}
```

```c
list_node *search_node(list *list_ad ,void *target_data ,int (*cmp)(void *A ,void *B))
{
    if (list_isok(list_ad))
    {
        //判断链表是否为空
        if (list_ad->head == NULL)
        {

            return NULL;

                return current;//找到了就返回当前节点

    }
    printf("链表无效\n");
    return NULL;
}
```

## 2. 栈

### 2.1 栈的结构

**栈是特殊的线性表，遵循一对一的关系**

物理结构上，可以由顺序表实现（数组），也可以由链式表实现（单向链表）

**栈限制在一边进行插入操作和删除操作：**

*   **允许操作的一端为栈顶**
*   **固定的一端为栈底**

**栈中没有任何元素时称之为空栈**

**栈中元素的进出顺序为：先进后出-FILO**

下面以单向链表的形式

栈节点

```c
//栈节点结构体
typedef struct stack_node

    //数据
    void *data;
    //下一个节点
    struct stack_node *next;
```

#### 2.1.1 入栈

```c
void push(stack *stack_ad ,void *data_ad)
{
    if (stack_isok(stack_ad))
    {
        //创建一个新的栈节点
        stack_node *new = create_stacknode(data_ad);
        //如果当前栈为空
        if (stack_ad->top == NULL)
        {
            stack_ad->top = new;
        }
        else//非空
        {
            //新节点的next = 旧栈顶
            new->next = stack_ad->top;
            //更新栈顶
            stack_ad->top = new;
        }
        //计数+1
        stack_ad->size ++;

        printf("入栈成功\n");
        return ;
    }
    printf("栈无效\n");
}

```

#### 2.1.2 出栈

```c
void *pop(stack *stack_ad)
{
    if (stack_isok(stack_ad))
    {
        //判断当前栈是否为空
        if (stack_ad->top == NULL)
        {
            printf("当前栈中无节点\n");
            return NULL;
        }
        else
        {
            //获取出栈数据
            stack_node *temp = stack_ad->top;
            void *data = temp->data;
            //更新栈顶
            stack_ad->top = stack_ad->top->next;
            //释放出栈元素
            free(temp);
            temp = NULL;

            //计数-1
            stack_ad->size --;

            printf("出栈成功\n");
            return data;
        }
        
    }
    printf("栈无效\n");
}
```
